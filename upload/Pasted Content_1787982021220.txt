### 2. Search the Verbatim Tier

Add a `search_verbatim()` method:

```python
def search_verbatim(self, query: str, user_id: str, k: int = 5):
    # BM25 ranking via FTS5
    bm25_results = self._execute(
        """SELECT rowid, rank FROM verbatim_fts 
           WHERE verbatim_fts MATCH ? AND user_id = ? 
           ORDER BY rank LIMIT ?""",
        (query, user_id, k * 2)
    ).fetchall()
    
    # Dense reranking of BM25 candidates
    query_vec = self.embedder.encode(query)
    scored = []
    for rowid, bm25_score in bm25_results:
        vec_row = self._execute(
            "SELECT vec FROM verbatim_vectors WHERE chunk_id = ?",
            (rowid,)
        ).fetchone()
        if vec_row:
            vec = np.frombuffer(vec_row[0], dtype=np.int8)
            sim = cosine_sim(query_vec, vec)
            scored.append((rowid, sim))
    
    scored.sort(key=lambda x: x[1], reverse=True)
    top_ids = [s[0] for s in scored[:k]]
    
    # Fetch full text
    if not top_ids:
        return []
    placeholders = ",".join("?" * len(top_ids))
    rows = self._execute(
        f"SELECT text, session_id, step_id FROM verbatim_chunks WHERE id IN ({placeholders})",
        tuple(top_ids)
    ).fetchall()
    return [dict(text=r[0], session_id=r[1], step_id=r[2]) for r in rows]
```

Add a `VerbatimAnswerer` that extracts answers without LLM:

```python
class VerbatimAnswerer:
    def __init__(self):
        # Reverse patterns: given a question, extract answer from chunk
        self.patterns = [
            # "What restaurant?" → "I loved the pasta at Italian Garden"
            (r"(?:loved|liked|enjoyed|adored).*?(?:the|at)\s+([A-Z][\w\s]+)", "restaurant"),
            # "What is your pet's name?" → "my dog is named Charlie"
            (r"(?:dog|cat|pet).*?named\s+([A-Z]\w+)", "pet"),
            # "Where do you work?" → "I work at Google"
            (r"(?:work|job).*?(?:at|for)\s+([A-Z][\w\s]+)", "work"),
        ]
    
    def extract(self, chunk: str, question: str) -> str:
        q = question.lower()
        for pattern, category in self.patterns:
            if category in q or any(w in q for w in category.split()):
                m = re.search(pattern, chunk, re.I)
                if m:
                    return m.group(1).strip()
        
        # Fallback: token overlap scoring
        return self._token_overlap_extract(chunk, question)
    
    def _token_overlap_extract(self, chunk: str, question: str) -> str:
        q_tokens = set(question.lower().split()) - {"the", "a", "what", "is", "did", "where", "who"}
        words = chunk.split()
        best_span, best_score = "", 0
        
        for length in range(1, 6):
            for i in range(len(words) - length + 1):
                span = " ".join(words[i:i+length])
                span_tokens = set(span.lower().split())
                score = len(span_tokens & q_tokens)
                if score > best_score:
                    best_score = score
                    best_span = span
        
        return best_span if best_score >= 1 else ""
```

### 5. Wire It Into the Reader

```python
def _context_block(self, query: str, user_id: str, **kw) -> str:
    plan = self._plan(query)
    
    if plan["mode"] == "verbatim":
        chunks = self.store.search_verbatim(query, user_id, k=3)
        if chunks:
            answers = []
            for c in chunks:
                ans = self.verbatim_answerer.extract(c["text"], query)
                if ans:
                    answers.append(f"ANSWER: {ans} (from: {c['text'][:80]}...)")
            if answers:
                return "\n".join(answers)
        # Fall through to structured if verbatim fails
    
    # EXISTING: structured path
    return self._structured_context_block(query, user_id, **kw)
```

### 2. Search the Verbatim Tier

Add a `search_verbatim()` method:

```python
def search_verbatim(self, query: str, user_id: str, k: int = 5):
    # BM25 ranking via FTS5
    bm25_results = self._execute(
        """SELECT rowid, rank FROM verbatim_fts 
           WHERE verbatim_fts MATCH ? AND user_id = ? 
           ORDER BY rank LIMIT ?""",
        (query, user_id, k * 2)
    ).fetchall()
    
    # Dense reranking of BM25 candidates
    query_vec = self.embedder.encode(query)
    scored = []
    for rowid, bm25_score in bm25_results:
        vec_row = self._execute(
            "SELECT vec FROM verbatim_vectors WHERE chunk_id = ?",
            (rowid,)
        ).fetchone()
        if vec_row:
            vec = np.frombuffer(vec_row[0], dtype=np.int8)
            sim = cosine_sim(query_vec, vec)
            scored.append((rowid, sim))
    
    scored.sort(key=lambda x: x[1], reverse=True)
    top_ids = [s[0] for s in scored[:k]]
    
    # Fetch full text
    if not top_ids:
        return []
    placeholders = ",".join("?" * len(top_ids))
    rows = self._execute(
        f"SELECT text, session_id, step_id FROM verbatim_chunks WHERE id IN ({placeholders})",
        tuple(top_ids)
    ).fetchall()
    return [dict(text=r[0], session_id=r[1], step_id=r[2]) for r in rows]
```

### 4. Answer Extraction from Verbatim Chunks

Add a `VerbatimAnswerer` that extracts answers without LLM:

```python
class VerbatimAnswerer:
    def __init__(self):
        # Reverse patterns: given a question, extract answer from chunk
        self.patterns = [
            # "What restaurant?" → "I loved the pasta at Italian Garden"
            (r"(?:loved|liked|enjoyed|adored).*?(?:the|at)\s+([A-Z][\w\s]+)", "restaurant"),
            # "What is your pet's name?" → "my dog is named Charlie"
            (r"(?:dog|cat|pet).*?named\s+([A-Z]\w+)", "pet"),
            # "Where do you work?" → "I work at Google"
            (r"(?:work|job).*?(?:at|for)\s+([A-Z][\w\s]+)", "work"),
        ]
    
    def extract(self, chunk: str, question: str) -> str:
        q = question.lower()
        for pattern, category in self.patterns:
            if category in q or any(w in q for w in category.split()):
                m = re.search(pattern, chunk, re.I)
                if m:
                    return m.group(1).strip()
        
        # Fallback: token overlap scoring
        return self._token_overlap_extract(chunk, question)
    
    def _token_overlap_extract(self, chunk: str, question: str) -> str:
        q_tokens = set(question.lower().split()) - {"the", "a", "what", "is", "did", "where", "who"}
        words = chunk.split()
        best_span, best_score = "", 0
        
        for length in range(1, 6):
            for i in range(len(words) - length + 1):
                span = " ".join(words[i:i+length])
                span_tokens = set(span.lower().split())
                score = len(span_tokens & q_tokens)
                if score > best_score:
                    best_score = score
                    best_span = span
        
        return best_span if best_score >= 1 else ""
```
