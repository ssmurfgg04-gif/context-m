     1→"""The μ=0 extraction pattern library.
     2→
     3→High-precision syntactic patterns over Subject-Relation-Value triples:
     4→first-person (user), third-person (entities), and assistant-turn
     5→(second-person) forms, with temporal anchors resolved by bridge.dates.
     6→Zero LLM calls — this is the deterministic perception layer that makes
     7→BEAM-honest μ=0 ingest possible while competitors burn LLM extraction
     8→at write time.
     9→"""

    54→@dataclass
    55→class Candidate:
    56→    subject: str
    57→    relation: str
    58→    value: str
    59→    confidence: float
    60→    pattern: str
    61→    span: tuple[int, int] = (0, 0)
    62→    valid_from: str | None = None
    63→    valid_to: str | None = None
    64→    retraction: bool = False
    65→    note: str = ""
    66→
    67→
    68→@dataclass
    69→class ExtractionContext:
    70→    user_id: str = "default"
    71→    agent_id: str | None = None
    72→    run_id: str | None = None
    73→    ts: datetime | None = None
    74→    speaker: str = "user"          # user | assistant | system
    75→    subject_name: str | None = None  # learned canonical name of the user
    76→    lexicon: set[str] = field(default_factory=set)
    77→
    78→    @property
    79→    def subject(self) -> str:
    80→        return self.subject_name or f"user:{self.user_id}"

    82→
    83→def clean_value(v: str) -> str:
    84→    v = v.strip().strip('"\',.!?;:')
    85→    v = re.sub(r"^(?:the|a|an)\s+", "", v, flags=re.I)
    86→    v = re.sub(r"\s+", " ", v)
    87→    v = v.strip().rstrip(".,!?;:")
    88→    return v.strip()
    89→
    90→
    91→def date_in(text: str, ts: datetime | None) -> str | None:
    92→    if ts is None:
    93→        return None
    94→    ds = find_dates(text, ts)
    95→    return ds[0]["iso"] if ds else None

    98→# ---------------------------------------------------------------- patterns
    99→# Each entry: (name, regex, handler(match, ctx, sent_span) -> list[Candidate])
   100→# m.group("val") etc. Handlers return candidates with subject placeholder
   101→# "SELF" resolved by the extractor to ctx.subject.
   102→
   103→PATTERNS: list[tuple[str, re.Pattern, object]] = []
   104→
   105→
   106→def pattern(name: str, rx: str):
   107→    compiled = re.compile(rx, re.I | re.M)
   108→
   109→    def deco(fn):
   110→        PATTERNS.append((name, compiled, fn))
   111→        return fn
   112→
   113→    return deco

   117→@pattern("name_intro", rf"\bmy name is\s+(?P<val>{NAME})")
   118→def _name(m, ctx, sp, ts, sent):
   119→    full = clean_value(m.group("val"))
   120→    out = []
   121→    if ctx.subject_name is None or ctx.subject_name.lower() != full.lower():
   122→        out.append(Candidate("SELF", "name", full, 0.95, "name_intro", (sp[0] + m.start(), sp[0] + m.end())))
   123→    parts = full.split()
   124→    if len(parts) > 1 and len(parts[0]) > 2:
   125→        out.append(Candidate(full, "alias", parts[0], 0.9, "name_intro_alias"))
   126→    return out

   540→# --- generic event with date --------------------------------------------------
   541→EVENT_START = re.compile(r"\b(?:i|we)\s+(?P<vp>[a-z][a-z' -]{2,60})", re.I)
   542→
   543→
   544→import re as _re2
   545→_DATE_GATE = _re2.compile(
   546→    r"\d|jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec|yesterday|today|"
   547→    r"tomorrow|last|ago", _re2.I)
   548→
   549→
   550→def extract_events(sent: str, sp: tuple[int, int], ts: datetime | None,
   551→                   ctx: ExtractionContext) -> list[Candidate]:
   552→    if ts is None or not _DATE_GATE.search(sent):
   553→        return []
   554→    dates = find_dates(sent, ts)
   555→    if not dates:
   556→        return []
   557→    m = EVENT_START.search(sent)
   558→    if not m:
   559→        return []
   560→    vp = m.group("vp").strip()
   561→    # strip a trailing date surface from the verb phrase
   562→    for d in dates:
   563→        surf = d["surface"].strip()
   564→        if surf and surf.lower() in vp.lower():
   565→            vp = re.sub(re.escape(surf), "", vp, flags=re.I).strip()
   566→    _MONTHS = "january|february|march|april|may|june|july|august|september|october|november|december"
   567→    for _ in range(3):
   568→        vp = re.sub(r"\s+(?:on|in|at|since|ago|back|last|this|next)\s*$", "", vp, flags=re.I).strip()
   569→        vp = re.sub(rf"\s+(?:{_MONTHS})\s*$", "", vp, flags=re.I).strip()
   570→        vp = re.sub(r"\s+\d{1,4}(?:st|nd|rd|th)?\s*$", "", vp).strip()
   571→        vp = re.sub(r"\s+(?:week|month|year|day)s?\s*$", "", vp, flags=re.I).strip()
   572→    vp = re.sub(r"\s+", " ", vp)
   573→    if len(vp) < 4 or vp.split()[0] in ("will", "would", "hope", "want", "plan"):
   574→        return []
   575→    # conversational fluff is not a memorable life event
   576→    _JUNK = re.compile(
   577→        r"^(?:read|reads|watch|watches|watched|see|saw|seen|hear|heard|"
   578→        r"think|thinks|thought|look|looked|talk|talked|remember|"
   579→        r"was|were|am|be|been)\b", re.I)
   580→    if _JUNK.match(vp):
   581→        return []
   582→    return [Candidate("SELF", "event", vp, 0.7, "event",
   583→                      valid_from=dates[0]["iso"],
   584→                      span=(sp[0] + m.start(), sp[0] + m.end()))]

   661→# --- section-aware kinship extraction (BEAM-10M user_relationships) -----------
   662→# The previous `profile_relative` pattern emitted every kinship bullet as
   663→# `related_to` because it didn't know which section header the bullet was
   664→# under. The fix is a multi-line pattern that captures the section header
   665→# + bullet as one regex, so the relation can be derived from the header
   666→# in the same pass.
   667→#
   668→# BEAM-10M user_relationships section looks like:
   669→#   PARENTS & GUARDIANS:
   670→#   • Alicia (female, age 80)
   671→#   • John (male, age 82)
   672→#   ROMANTIC PARTNER:
   673→#   • Chris (35, software engineer)
   674→#   CHILDREN:
   675→#   • Brittany (12)
   676→#   • Maria (8)
   677→#   SIBLINGS:
   678→#   • Sam (28, lawyer)
   679→#   FRIENDS:
   680→#   • Bob (40, teacher)
   681→#
   682→# Section→relation map (BEAM-10M canonical headers):
   683→_KINSHIP_SECTIONS = {
   684→    "PARENTS & GUARDIANS": "parent",
   685→    "PARENTS": "parent",
   686→    "GUARDIANS": "parent",
   687→    "ROMANTIC PARTNER": "partner",
   688→    "ROMANTIC PARTNERS": "partner",
   689→    "PARTNER": "partner",
   690→    "PARTNERS": "partner",
   691→    "SPOUSE": "spouse",
   692→    "CHILDREN": "child",
   693→    "CHILD": "child",
   694→    "SIBLINGS": "sibling",
   695→    "SIBLING": "sibling",
   696→    "FRIENDS": "friend",
   697→    "FRIEND": "friend",
   698→    "COLLEAGUES": "colleague",
   699→    "COLLEAGUE": "colleague",
   700→    "COWORKERS": "colleague",
   701→    "EXTENDED FAMILY": "family",
   702→    "GRANDPARENTS": "parent",  # grandparent is a parent role
   703→    "GRANDCHILDREN": "child",
   704→    "NEPHEWS & NIECES": "sibling",  # sibling lineage
   705→    "IN-LAWS": "family",
   706→}
   707→# Build one big alternation of section headers (longest first so
   708→# "PARENTS & GUARDIANS" beats "PARENTS")
   709→_KINSHIP_HEADERS = sorted(_KINSHIP_SECTIONS.keys(), key=len, reverse=True)
   710→_KINSHIP_HEADERS_RE = "|".join(
   711→    re.escape(h).replace(r"\ ", r"\s+") for h in _KINSHIP_HEADERS)
   712→
   713→
   714→@pattern("profile_kinship_section",
   715→         rf"(?P<section>{_KINSHIP_HEADERS_RE})\s*:\s*\n"
   716→         rf"(?P<bullets>(?:\s*•\s+{NAME}\s*\([^)]*\)\s*\n?)+)")
   717→def _profile_kinship_section(m, ctx, sp, ts, sent):
   718→    """Section-aware kinship extraction.
   719→
   720→    Matches a whole section header + bullet block, emits one Candidate
   721→    per bullet with the section-derived relation. This replaces the
   722→    lossy `related_to` fallback that the single-line `profile_relative`
   723→    pattern was emitting.
   724→    """
   725→    section_raw = m.group("section")
   726→    # normalize header to canonical form (collapse whitespace, uppercase)
   727→    section_key = re.sub(r"\s+", " ", section_raw).strip().upper()
   728→    relation = _KINSHIP_SECTIONS.get(section_key)
   729→    if relation is None:
   730→        # try matching by prefix (e.g. "PARENTS & GUARDIANS" might have
   731→        # spacing differences)
   732→        for h in _KINSHIP_HEADERS:
   733→            if h in section_key:
   734→                relation = _KINSHIP_SECTIONS[h]
   735→                break
   736→    if relation is None:
   737→        return []  # unknown section — leave to single-line pattern
   738→    bullets = m.group("bullets")
   739→    target = ctx.subject_name or "SELF"
   740→    out: list[Candidate] = []
   741→    # iterate bullets within the section
   742→    for bm in re.finditer(rf"•\s+(?P<val>{NAME})\s*\([^)]*\)", bullets):
   743→        v = clean_value(bm.group("val"))
   744→        if v and len(v) >= 2:
   745→            out.append(Candidate(target, relation, v, 0.85,
   746→                                 "profile_kinship_section"))
   747→    return out
