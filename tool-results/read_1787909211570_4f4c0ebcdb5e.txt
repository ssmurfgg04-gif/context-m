    52→# ------------------------------------------------------------------ ratelimit
    53→class TokenBucket:
    54→    def __init__(self, rate: float, burst: int) -> None:
    55→        self.rate = rate
    56→        self.burst = burst
    57→        self._tokens: dict[str, list[float]] = {}
    58→        self._lock = threading.Lock()
    59→
    60→    def allow(self, key: str) -> bool:
    61→        now = time.monotonic()
    62→        with self._lock:
    63→            tokens, ts = self._tokens.get(key, [float(self.burst), now])
    64→            refill = (now - ts) * self.rate
    65→            tokens = min(float(self.burst), tokens + refill)
    66→            if tokens < 1.0:
    67→                self._tokens[key] = [tokens, now]
    68→                return False
    69→            self._tokens[key] = [tokens - 1.0, now]
    70→            return True

    75→    def op(summary: str, params: list | None = None, body: bool = False,

    77→        d: dict = {"summary": summary, "responses": responses or

    79→                    "401": {"description": "invalid or missing API key"},
    80→                    "403": {"description": "role not permitted"},
    81→                    "429": {"description": "rate limited"}}}
    82→        tag = {"$ref": "#/components/securitySchemes/bearer"}
    83→        if roles:
    84→            d["description"] = f"roles: {', '.join(roles)}"
    85→        d["security"] = [tag]
    86→        if body:
    87→            d["requestBody"] = {"required": True, "content": {
    88→                "application/json": {"schema": {"type": "object"}}}}
    89→        return d

   183→def build_handler(state: FabricState):
   184→
   185→    class Handler(BaseHTTPRequestHandler):
   186→        server_version = "context-m/1.0"
   187→        protocol_version = "HTTP/1.1"
   188→
   189→        # -------------------------------------------------------- plumbing
   190→        def log_message(self, fmt, *args):  # quiet access log
   191→            pass
   192→
   193→        def _send(self, code: int, payload, content_type="application/json"):
   194→            body = (payload if isinstance(payload, (bytes, bytearray))
   195→                    else json.dumps(payload, default=str).encode())
   196→            self.send_response(code)
   197→            self.send_header("Content-Type", content_type)
   198→            self.send_header("Content-Length", str(len(body)))
   199→            self.end_headers()
   200→            try:
   201→                self.wfile.write(body)
   202→            except (BrokenPipeError, ConnectionResetError):
   203→                pass
   204→
   205→        def _body(self) -> dict:
   206→            length = int(self.headers.get("Content-Length") or 0)
   207→            if length > MAX_BODY:
   208→                raise ValueError("body too large")
   209→            if not length:
   210→                return {}
   211→            raw = self.rfile.read(length)
   212→            try:
   213→                return json.loads(raw or b"{}")
   214→            except json.JSONDecodeError:
   215→                raise ValueError("invalid JSON body")
   216→
   217→        def _auth(self, action: str) -> dict:
   218→            hdr = self.headers.get("Authorization") or ""
   219→            if not hdr.startswith("Bearer "):
   220→                REGISTRY.inc("contextm_http_requests_total",
   221→                             {"code": "401"})
   222→                self._send(401, {"error": "missing bearer token"})
   223→                return {}
   224→            key = hdr[7:].strip()
   225→            meta = state.keys.verify(key)
   226→            if meta is None:
   227→                state.memory.audit_log.log("auth.failure",
   228→                                       actor=key[:16] + "…",
   229→                                       outcome="invalid_key")
   230→                REGISTRY.inc("contextm_http_requests_total",
   231→                             {"code": "401"})
   232→                self._send(401, {"error": "invalid or revoked key"})
   233→                return {}
   234→            if not state.bucket.allow(key):
   235→                REGISTRY.inc("contextm_http_requests_total",
   236→                             {"code": "429"})
   237→                self._send(429, {"error": "rate limit exceeded"})
   238→                return {}
   239→            try:
   240→                authorize(meta, action)
   241→            except RBACError as e:
   242→                state.memory.audit_log.log(action, actor=meta.get("label")
   243→                                       or meta.get("id", "key"),
   244→                                       role=meta.get("role"),
   245→                                       outcome="denied")
   246→                REGISTRY.inc("contextm_http_requests_total",
   247→                             {"code": "403"})
   248→                self._send(403, {"error": str(e)})
   249→                return {}
   250→            REGISTRY.inc("contextm_http_requests_total", {"code": "200"})
   251→            return meta
   252→

   356→        def do_POST(self):
   357→            path = self.path.split("?")[0].rstrip("/")
   358→            t0 = time.monotonic()
   359→            try:
   360→                body = self._body()
   361→            except ValueError as e:
   362→                return self._send(400, {"error": str(e)})
   363→
   364→            routes = {
   365→                "/v1/add": ("memory.add", self._h_add),
   366→                "/v1/search": ("memory.search", self._h_search),
   367→                "/v1/keys": ("keys.create", self._h_key_create),
   368→                "/v1/snapshot": ("governance.snapshot", self._h_snapshot),
   369→                "/v1/restore": ("governance.restore", self._h_restore),
   370→                "/v1/erase": ("governance.erase", self._h_erase),
   371→                "/v1/retention": ("governance.retention", self._h_retention),
   372→                "/v1/state_at": ("governance.pitr", self._h_state_at),
   373→                "/v1/sparql": ("sparql.query", self._h_sparql),
   374→                "/v1/export": ("memory.export", self._h_export),
   375→                "/v1/consolidate": ("governance.consolidate",
   376→                                       self._h_consolidate),
   377→                "/v1/chaos": ("memory.chaos_ingest", self._h_chaos),
   378→            }
   379→            if path not in routes:
   380→                return self._send(404, {"error": f"no route {path}"})
   381→            action, fn = routes[path]
   382→            meta = self._auth(action)
   383→            if not meta:
   384→                return
   385→            actor = meta.get("label") or meta.get("id", "key")
   386→            try:
   387→                out = fn(body, actor, meta)
   388→                REGISTRY.observe("contextm_http_request_seconds",
   389→                                 time.monotonic() - t0)
   390→                return self._send(200, out)
   391→            except RBACError as e:
   392→                return self._send(403, {"error": str(e)})
   393→            except Exception as e:  # noqa: BLE001
   394→                REGISTRY.inc("contextm_http_requests_total",
   395→                             {"code": "500"})
   396→                state.memory.audit_log.log(action, actor=actor,
   397→                                       role=meta.get("role"),
   398→                                       outcome="error",
   399→                                       meta={"error": str(e)[:200]})
   400→                return self._send(500, {"error": str(e)[:500]})

   415→                    return

   421→                                       outcome="revoked" if ok else "missing")

   505→        def _h_sparql(self, body, actor, meta):
   506→            """Inline SPARQL endpoint — auth'd, shares one Memory.
   507→
   508→            GET  /v1/sparql?query=SELECT...
   509→            POST /v1/sparql  {"query": "SELECT ..."}
   510→            """
   511→            # body is {} for GET — parse query string instead
   512→            query = (body.get("query") if body else None) or ""
   513→            if not query:
   514→                from urllib.parse import parse_qs, urlparse
   515→                q = parse_qs(urlparse(self.path).query)
   516→                query = q.get("query", [""])[0]
   517→            if not query:
   518→                return {"error": "missing 'query' parameter"}
   519→            try:
   520→                from context_m.server.sparql import (
   521→                    execute_sparql, edge_triples)
   522→                with state.lock:
   523→                    fact_objs = state.memory.store.query_facts(
   524→                        active=True,
   525→                        user_id=body.get("user_id") if body else None)
   526→                    facts = [(f.subject, f.relation, f.value)
   527→                             for f in fact_objs]
   528→                    edges = edge_triples(state.memory,
   529→                                          user_id=body.get("user_id")
   530→                                          if body else None)
   531→                    arena = getattr(state.memory, "blob_arena", None)
   532→                    blob_resolver = None
   533→                    if arena is not None:
   534→                        def _resolve(_s, _r, val):
   535→                            from context_m.trace.blob_arena import \
   536→                                get_chunk_text
   537→                            return get_chunk_text(state.memory.store,
   538→                                                   arena, val)
   539→                        blob_resolver = _resolve
   540→                out = execute_sparql(query, facts,
   541→                                       user_id=body.get("user_id")
   542→                                       if body else None,
   543→                                       edge_triples=edges,
   544→                                       blob_resolver=blob_resolver)
   545→                state.memory.audit_log.log(
   546→                    "sparql.query", actor=actor, role=meta.get("role"),
   547→                    meta={"n_results": out.get("n_results", 0),
   548→                           "query_head": query[:80]})
   549→                return out
   550→            except ValueError as e:
   551→                return {"error": str(e)}
   552→            except Exception as e:  # noqa: BLE001
   553→                return {"error": f"server: {e}"}

   555→        def _h_export(self, body, actor, meta):
   556→            """Export facts via the swappable decoder.
   557→
   558→            GET /v1/export?format=rdf|json|datalog|llm_prompt&user_id=X
   559→            """
   560→            from urllib.parse import parse_qs, urlparse
   561→            q = parse_qs(urlparse(self.path).query)
   562→            fmt = (q.get("format") or ["llm_prompt"])[0]
   563→            user_id = (q.get("user_id") or [None])[0]
   564→            from context_m.bridge.decoders import get_decoder
   565→            try:
   566→                decoder = get_decoder(fmt)
   567→            except ValueError as e:
   568→                return {"error": str(e)}
   569→            with state.lock:
   570→                facts = state.memory.store.query_facts(
   571→                    active=True, user_id=user_id)
   572→                # score by recency for export ordering
   573→                scores = {f.id: 1.0 for f in facts}
   574→                out = decoder.render(
   575→                    query="", intent="export", facts=list(facts),
   576→                    scores=scores, notes=None,
   577→                    store=state.memory.store)
   578→            state.memory.audit_log.log(
   579→                "memory.export", actor=actor, role=meta.get("role"),
   580→                resource=user_id or "default",
   581→                meta={"format": fmt, "n_facts": len(facts)})
   582→            # for json decoder, return parsed; otherwise return text
   583→            if fmt == "json":
   584→                try:
   585→                    return json.loads(out)
   586→                except json.JSONDecodeError:
   587→                    pass
   588→            return {"format": fmt, "n_facts": len(facts),
   589→                    "content": out}

   612→            """Zero-config auto-ingest (EAM chaos mode).
   613→
   614→            POST /v1/chaos {"texts": ["...", ...], "user_id": "default"}
   615→            POST /v1/chaos {"text": "...", "user_id": "default"}

   621→            if not texts:

   623→            user_id = body.get("user_id", "default")
   624→            with state.lock:
   625→                out = chaos_ingest(state.memory, texts, user_id=user_id,
   626→                                   agent_id=body.get("agent_id"),
   627→                                   run_id=body.get("run_id"))
   628→            state.memory.audit_log.log(
   629→                "memory.chaos_ingest", actor=actor, role=meta.get("role"),
   630→                resource=user_id,
   631→                meta={"n_facts": out.get("stats", {}).get(
   632→                    "facts_inserted", 0)})
   633→            return out

   638→# ------------------------------------------------------------------ launch
   639→def serve(memory: Memory | None = None, host: str = "0.0.0.0",
   640→          port: int = 8900, *, sparql_port: int | None = None,
   641→          sparql_host: str = "0.0.0.0",
   642→          sparql_user_id: str | None = None) -> ThreadingHTTPServer:
   643→    if memory is None:
   644→        memory = Memory(Config.from_env())
   645→    state = FabricState(memory)
   646→    httpd = ThreadingHTTPServer((host, port), build_handler(state))
   647→    httpd.daemon_threads = True
   648→    # optionally co-host a SPARQL endpoint sharing the same Memory instance
   649→    if sparql_port:
   650→        from context_m.server.sparql import SparqlServer
   651→        sparql = SparqlServer(memory, host=sparql_host, port=sparql_port,
   652→                               user_id=sparql_user_id)
   653→        sparql.start_background()
   654→        httpd.sparql = sparql  # type: ignore[attr-defined]
   655→    return httpd
   656→
   657→
   658→def main() -> None:
   659→    ap = argparse.ArgumentParser(prog="contextm-serve",
   660→                                 description="Context-M REST API server")
   661→    ap.add_argument("--db", default=":memory:", help="SQLite path")
   662→    ap.add_argument("--host", default="0.0.0.0")
   663→    ap.add_argument("--port", type=int, default=8900)
   664→    ap.add_argument("--pii", default=None,
   665→                    help="off|redact|block|tag (default: config/env)")
   666→    ap.add_argument("--admin-key", default=None,
   667→                    help="create this admin API key at boot (dev convenience)")
   668→    ap.add_argument("--sparql-port", type=int, default=None,
   669→                    help="co-host a SPARQL endpoint on this port "
   670→                         "(shares one Memory instance with the REST API)")
   671→    ap.add_argument("--sparql-host", default="0.0.0.0",
   672→                    help="bind SPARQL endpoint to this host")
   673→    ap.add_argument("--sparql-user-id", default=None,
   674→                    help="scope SPARQL queries to a single user")
   675→    args = ap.parse_args()
   676→
   677→    cfg = Config.from_env(db_path=args.db)
   678→    if args.pii:
   679→        cfg.pii_mode = args.pii
   680→    memory = Memory(cfg)
   681→    if args.admin_key:
   682→        meta = memory.keys.create("admin", label="boot-admin")
   683→        print(f"[context-m] admin API key (save now, shown once): {meta['key']}")
   684→    httpd = serve(memory, args.host, args.port,
   685→                  sparql_port=args.sparql_port,
   686→                  sparql_host=args.sparql_host,
   687→                  sparql_user_id=args.sparql_user_id)
   688→    print(f"[context-m] REST API on http://{args.host}:{args.port}"
   689→          f"  (db={args.db}, pii={cfg.pii_mode}, "
   690→          f"encrypt={cfg.encryption_at_rest})")
   691→    if args.sparql_port:
   692→        print(f"[context-m] SPARQL endpoint on http://{args.sparql_host}:"
   693→              f"{args.sparql_port}/  (co-hosted, shares Memory)")
   694→
   695→    # graceful shutdown on SIGTERM/SIGINT — drain in-flight requests
   696→    def _shutdown(signum, frame):
   697→        print(f"\n[context-m] received signal {signum} — shutting down...")
   698→        sparql = getattr(httpd, "sparql", None)
   699→        if sparql is not None:
   700→            sparql.stop()
   701→        httpd.shutdown()  # unblocks serve_forever()
   702→    signal.signal(signal.SIGTERM, _shutdown)
   703→    signal.signal(signal.SIGINT, _shutdown)
   704→    try:
   705→        httpd.serve_forever()
   706→    except KeyboardInterrupt:
   707→        pass
   708→    finally:
   709→        memory.close()
   710→
   711→
   712→if __name__ == "__main__":
   713→    main()
