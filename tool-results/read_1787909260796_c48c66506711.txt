    77→class _SafeCursor:
    78→    """Cursor-like object over eagerly-materialized rows (thread-safe)."""
    79→
    80→    def __init__(self, rows, rowcount, lastrowid, description) -> None:
    81→        self._rows = rows
    82→        self._iter = iter(rows)
    83→        self.rowcount = rowcount
    84→        self.lastrowid = lastrowid
    85→        self.description = description
    86→        self.arraysize = 1
    87→
    88→    def fetchone(self):
    89→        try:
    90→            return next(self._iter)
    91→        except StopIteration:
    92→            return None
    93→
    94→    def fetchall(self):
    95→        rest = list(self._iter)
    96→        self._iter = iter([])
    97→        return rest
    98→
    99→    def fetchmany(self, size=None):
   100→        out = []
   101→        for _ in range(size or 1):
   102→            try:
   103→                out.append(next(self._iter))
   104→            except StopIteration:
   105→                break
   106→        return out
   107→
   108→    def __iter__(self):
   109→        return self
   110→
   111→    def __next__(self):
   112→        return next(self._iter)
   113→
   114→    def close(self) -> None:
   115→        pass

   118→class SafeConnection:
   119→    """Serializes every statement on one SQLite connection.
   120→
   121→    SQLite connections are not safe for concurrent cursor use even with
   122→    ``check_same_thread=False`` — interleaved commit/iterate produces
   123→    InterfaceError('bad parameter or other API misuse'). This wrapper
   124→    holds an RLock across execute+materialize and across commit, making
   125→    the whole TraceStore safe under multi-threaded load (REST server,
   126→    concurrent writers). Eager materialization keeps semantics: callers
   127→    only use fetchone/fetchall/iteration/rowcount/lastrowid.
   128→    """
   129→
   130→    def __init__(self, conn) -> None:
   131→        self._conn = conn
   132→        self._lock = threading.RLock()
   133→
   134→    def execute(self, sql, params=()):
   135→        with self._lock:
   136→            cur = self._conn.execute(sql, params)
   137→            try:
   138→                rows = cur.fetchall()
   139→            except Exception:
   140→                rows = []
   141→            rc, lrid, desc = cur.rowcount, cur.lastrowid, cur.description
   142→            cur.close()
   143→            return _SafeCursor(rows, rc, lrid, desc)
   144→
   145→    def executemany(self, sql, seq):
   146→        with self._lock:
   147→            cur = self._conn.executemany(sql, seq)
   148→            rc = cur.rowcount
   149→            cur.close()
   150→            return _SafeCursor([], rc, None, None)
   151→
   152→    def executescript(self, script):
   153→        with self._lock:
   154→            return self._conn.executescript(script)
   155→
   156→    def commit(self):
   157→        with self._lock:
   158→            self._conn.commit()
   159→
   160→    def rollback(self):
   161→        with self._lock:
   162→            self._conn.rollback()
   163→
   164→    def close(self):
   165→        with self._lock:
   166→            self._conn.close()
   167→
   168→    @property
   169→    def row_factory(self):
   170→        return self._conn.row_factory
   171→
   172→    @row_factory.setter
   173→    def row_factory(self, v):
   174→        with self._lock:
   175→            self._conn.row_factory = v
   176→
   177→    @property
   178→    def in_transaction(self):
   179→        return self._conn.in_transaction

   182→class TraceStore:
   183→    def __init__(self, db_path: str = ":memory:", provider: HashProvider | None = None,
   184→                 wal_sync: str = "normal") -> None:
   185→        self.db_path = db_path
   186→        self.hasher = provider or HashProvider()
   187→        mem = db_path in (":memory:", None, "")
   188→        self.conn = SafeConnection(
   189→            sqlite3.connect(db_path or ":memory:", check_same_thread=False))
   190→        if not mem:
   191→            # Aeon-inspired crash-recoverable write path:
   192→            #   journal_mode=WAL — readers never block the writer, and a
   193→            #                       torn write rolls back cleanly on reopen.
   194→            #   synchronous      — NORMAL: commits survive process crash
   195→            #                       (SIGKILL) at full speed; FULL additionally
   196→            #                       survives OS/power loss at fsync cost.
   197→            self.conn.execute("PRAGMA journal_mode=WAL")
   198→            self.conn.execute(
   199→                "PRAGMA synchronous="
   200→                + ("FULL" if str(wal_sync).lower() == "full" else "NORMAL"))
   201→        self.conn.row_factory = sqlite3.Row
   202→        self.conn.executescript(SCHEMA)
   203→        self._ancestry_cache: dict[str, frozenset[str]] = {}
   204→        self._active_cache: dict[str, frozenset[str]] = {}
   205→        self._batching = False
   206→        self._ensure_genesis()
   207→
   208→    def checkpoint(self, mode: str = "TRUNCATE") -> None:
   209→        """Fold the WAL back into the main db file (shrink + fast reopen)."""
   210→        if self.db_path in (":memory:", None, ""):
   211→            return
   212→        try:
   213→            self.conn.execute(f"PRAGMA wal_checkpoint({mode})")
   214→        except Exception:
   215→            pass
   216→
   217→    def begin_batch(self) -> None:
   218→        self._batching = True
   219→
   220→    def end_batch(self) -> None:
   221→        self._batching = False
   222→        self.conn.commit()
   223→
   224→    def _maybe_commit(self) -> None:
   225→        if not self._batching:
   226→            self.conn.commit()
   227→
   228→    @property
   229→    def batching(self) -> bool:
   230→        return self._batching

   234→        row = self.conn.execute("SELECT v FROM kv WHERE k=?", (key,)).fetchone()

   238→        self.conn.execute(
   239→            "INSERT INTO kv(k, v) VALUES(?, ?) ON CONFLICT(k) DO UPDATE SET v=excluded.v",

   243→    def iter_kv(self, prefix: str = ""):
   244→        """Yield (key, value) pairs whose key starts with ``prefix``."""
   245→        cur = self.conn.execute(
   246→            "SELECT k, v FROM kv WHERE k LIKE ? ORDER BY k",
   247→            (prefix + "%",))
   248→        for row in cur:
   249→            yield row["k"], row["v"]

   252→        self.conn.execute("DELETE FROM kv WHERE k=?", (key,))

   255→    # -------------------------------------------------------------- genesis
   256→    def _ensure_genesis(self) -> None:
   257→        if not self.conn.execute("SELECT 1 FROM branches LIMIT 1").fetchone():
   258→            cid = new_id()
   259→            now = iso(datetime.utcnow().__class__.now() if False else datetime.utcnow()) if False else iso(datetime.now())  # noqa
   260→            chain = self.hasher.hash_text("genesis:" + cid)
   261→            self.conn.execute(
   262→                "INSERT INTO commits(id, parents, branch, message, ts, chain_hash) VALUES(?,?,?,?,?,?)",
   263→                (cid, "[]", "main", "genesis", now, chain))
   264→            self.conn.execute(
   265→                "INSERT INTO branches(name, head, created) VALUES(?,?,?)", ("main", cid, now))
   266→            self.kv_set("HEAD_BRANCH", "main")
   267→            self.kv_set("SCHEMA_VERSION", "1")

   269→    # -------------------------------------------------------------- chunks
   270→    def add_chunk(self, text: str, *, user_id: str = "default", agent_id: str | None = None,
   271→                  run_id: str | None = None, ts: datetime | str | None = None,
   272→                  source: str = "", chunk_id: str | None = None) -> str:
   273→        cid = chunk_id or new_id()
   274→        ts_s = iso(parse_ts(ts) or datetime.now(timezone.utc)) if ts else iso(datetime.now(timezone.utc))
   275→        self.conn.execute(
   276→            "INSERT OR REPLACE INTO chunks(id, text, user_id, agent_id, run_id, ts, source, hash, tokens) "
   277→            "VALUES(?,?,?,?,?,?,?,?,?)",
   278→            (cid, text, user_id, agent_id, run_id, ts_s, source,
   279→             self.hasher.hash_text(text), token_estimate(text)))
   280→        self._maybe_commit()
   281→        return cid

   305→    # ------------------------------------------------------------- commits
   306→    def create_commit(self, message: str = "", branch: str | None = None,
   307→                      parents: list[str] | None = None, n_facts: int = 0) -> str:
   308→        branch = branch or self.current_branch()
   309→        head = self.head(branch)
   310→        parents = parents if parents is not None else ([head] if head else [])
   311→        cid = new_id()
   312→        now = iso(datetime.now(timezone.utc))
   313→        parent_chains = [self.conn.execute(
   314→            "SELECT chain_hash FROM commits WHERE id=?", (p,)).fetchone() for p in parents]
   315→        chain = self.hasher.hash_json({
   316→            "commit": cid, "parents": parents, "message": message,
   317→            "ts": now,
   318→            "parent_chains": [r["chain_hash"] if r else "" for r in parent_chains],
   319→        })
   320→        self.conn.execute(
   321→            "INSERT INTO commits(id, parents, branch, message, ts, chain_hash, n_facts) VALUES(?,?,?,?,?,?,?)",
   322→            (cid, json.dumps(parents), branch, message, now, chain, n_facts))
   323→        if not parents:
   324→            self.conn.execute(
   325→                "INSERT OR REPLACE INTO branches(name, head, created) VALUES(?,?,?)",
   326→                (branch, cid, now))
   327→        else:
   328→            self.conn.execute("UPDATE branches SET head=? WHERE name=?", (cid, branch))
   329→        self._invalidate(branch)
   330→        self._maybe_commit()
   331→        return cid
   332→
   333→    def head(self, branch: str | None = None) -> str | None:
   334→        branch = branch or self.current_branch()
   335→        row = self.conn.execute("SELECT head FROM branches WHERE name=?", (branch,)).fetchone()
   336→        return row["head"] if row else None

   338→    def current_branch(self) -> str:
   339→        return self.kv_get("HEAD_BRANCH", "main") or "main"

   341→    def checkout(self, branch: str) -> None:
   342→        if not self.conn.execute("SELECT 1 FROM branches WHERE name=?", (branch,)).fetchone():
   343→            raise BranchError(f"unknown branch {branch!r}")
   344→        self.kv_set("HEAD_BRANCH", branch)
   345→
   346→    def create_branch(self, name: str, from_commit: str | None = None,
   347→                      switch: bool = True) -> str:
   348→        if self.conn.execute("SELECT 1 FROM branches WHERE name=?", (name,)).fetchone():
   349→            raise BranchError(f"branch {name!r} already exists")
   350→        base = from_commit or self.head() or self.head(self.current_branch())
   351→        if base is None:
   352→            raise BranchError("cannot branch from empty history")
   353→        self.conn.execute(
   354→            "INSERT INTO branches(name, head, created) VALUES(?,?,?)",
   355→            (name, base, iso(datetime.now(timezone.utc))))
   356→        if switch:
   357→            self.kv_set("HEAD_BRANCH", name)
   358→        self._maybe_commit()
   359→        return base

   361→    def branches(self) -> list[dict]:
   362→        rows = self.conn.execute(
   363→            "SELECT b.name, b.head, b.created, c.ts AS head_ts, c.message "
   364→            "FROM branches b LEFT JOIN commits c ON c.id=b.head ORDER BY b.created").fetchall()
   365→        return [dict(r) for r in rows]

   367→    def log(self, branch: str | None = None, limit: int = 50) -> list[dict]:
   368→        cid = self.head(branch or self.current_branch())
   369→        out: list[dict] = []
   370→        seen = set()
   371→        queue = [cid] if cid else []
   372→        while queue and len(out) < limit:
   373→            cur = queue.pop(0)
   374→            if not cur or cur in seen:
   375→                continue
   376→            seen.add(cur)
   377→            row = self.conn.execute("SELECT * FROM commits WHERE id=?", (cur,)).fetchone()
   378→            if not row:
   379→                continue
   380→            out.append(dict(row))
   381→            queue.extend(json.loads(row["parents"]))
   382→        return out

   384→    def ancestry(self, commit_id: str) -> frozenset[str]:
   385→        cached = self._ancestry_cache.get(commit_id)
   386→        if cached is not None:
   387→            return cached
   388→        seen: set[str] = set()
   389→        queue = [commit_id]
   390→        while queue:
   391→            cur = queue.pop()
   392→            if cur in seen:
   393→                continue
   394→            seen.add(cur)
   395→            row = self.conn.execute("SELECT parents FROM commits WHERE id=?", (cur,)).fetchone()
   396→            if row:
   397→                queue.extend(json.loads(row["parents"]))
   398→        result = frozenset(seen)
   399→        if len(self._ancestry_cache) < 64:
   400→            self._ancestry_cache[commit_id] = result
   401→        return result

   408→    def insert_fact(self, fact: Fact, commit_id: str | None = None) -> Fact:
   409→        if commit_id:
   410→            fact.birth_commit = commit_id
   411→        row = fact.to_row()
   412→        cols = ", ".join(row.keys())
   413→        ph = ", ".join("?" for _ in row)
   414→        self.conn.execute(f"INSERT INTO facts({cols}) VALUES({ph})", tuple(row.values()))
   415→        return fact

   417→    def insert_facts_bulk(self, facts: list[Fact], commit_id: str | None = None) -> int:
   418→        for f in facts:
   419→            if commit_id:
   420→                f.birth_commit = commit_id
   421→        rows = [f.to_row() for f in facts]
   422→        if not rows:
   423→            return 0
   424→        cols = ", ".join(rows[0].keys())
   425→        ph = ", ".join("?" for _ in rows[0])
   426→        self.conn.executemany(f"INSERT INTO facts({cols}) VALUES({ph})",
   427→                              [tuple(r.values()) for r in rows])
   428→        return len(rows)

   434→    def get_facts(self, ids: list[str]) -> list[Fact]:
   435→        if not ids:
   436→            return []
   437→        out = []
   438→        for i in range(0, len(ids), 500):

   444→    def update_fact(self, fact_id: str, **fields) -> None:

   458→    def _fact_filters(self, user_id=None, agent_id=None, run_id=None, branch=None,
   459→                      active=True, include_quarantined=False, subject=None,
   460→                      relation=None, value=None, derived=None):
   461→        clauses, params = [], []
   462→        if active:
   463→            clauses.append("is_active=1")
   464→        if not include_quarantined:
   465→            clauses.append("quarantined=0")
   466→        if user_id is not None:
   467→            clauses.append("user_id=?"); params.append(user_id)
   468→        if agent_id is not None:
   469→            clauses.append("agent_id=?"); params.append(agent_id)
   470→        if run_id is not None:
   471→            clauses.append("run_id=?"); params.append(run_id)
   472→        if subject is not None:
   473→            clauses.append("subject=?"); params.append(subject)
   474→        if relation is not None:
   475→            clauses.append("relation=?"); params.append(relation)
   476→        if value is not None:
   477→            clauses.append("value=?"); params.append(value)
   478→        if derived is not None:
   479→            clauses.append("is_derived=?"); params.append(int(derived))
   480→        if branch is not None:
   481→            ids = self.active_ids(branch)
   482→            if not ids:
   483→                clauses.append("1=0")
   484→            else:
   485→                # membership filtering applied post-hoc for large sets
   486→                pass
   487→        return clauses, params, (branch if branch is not None else None)
   488→
   489→    def query_facts(self, *, subject=None, relation=None, value=None, user_id=None,
   490→                    agent_id=None, run_id=None, branch=None, active=True,
   491→                    include_quarantined=False, derived=None, order="valid_from",
   492→                    limit=None) -> list[Fact]:
   493→        clauses, params, branch_filter = self._fact_filters(
   494→            user_id, agent_id, run_id, branch, active, include_quarantined,
   495→            subject, relation, value, derived)
   496→        where = (" WHERE " + " AND ".join(clauses)) if clauses else ""
   497→        q = f"SELECT {FACT_COLUMNS} FROM facts{where} ORDER BY {order}"
   498→        if limit:
   499→            q += f" LIMIT {int(limit)}"
   500→        rows = self.conn.execute(q, params).fetchall()
   501→        facts = [Fact.from_row(dict(r)) for r in rows]
   502→        if branch_filter is not None:
   503→            ids = self.active_ids(branch_filter)
   504→            facts = [f for f in facts if f.id in ids]
   505→        return facts
   506→
   507→    def active_facts(self, **kw) -> list[Fact]:
   508→        return self.query_facts(**kw)

   510→    def history_of(self, subject: str, relation: str, user_id: str | None = None,
   511→                   include_inactive=True) -> list[Fact]:
   512→        clauses, params, _ = self._fact_filters(
   513→            user_id=user_id, active=not include_inactive)
   514→        extra = "subject=? AND relation=?" + ((" AND " + " AND ".join(clauses)) if clauses else "")
   515→        q = f"SELECT {FACT_COLUMNS} FROM facts WHERE {extra} ORDER BY valid_from, tx_from"
   516→        rows = self.conn.execute(q, (subject, relation, *params)).fetchall()
   517→        return [Fact.from_row(dict(r)) for r in rows]

   519→    def facts_about(self, entity: str, user_id: str | None = None,
   520→                    active: bool = True) -> list[Fact]:
   521→        """Facts where entity is subject OR value (1-hop associative recall)."""
   522→        clauses, params, _ = self._fact_filters(user_id=user_id, active=active)
   523→        extra = "(subject=? OR value=?)" + ((" AND " + " AND ".join(clauses)) if clauses else "")
   524→        q = f"SELECT {FACT_COLUMNS} FROM facts WHERE {extra}"
   525→        rows = self.conn.execute(q, (entity, entity, *params)).fetchall()
   526→        return [Fact.from_row(dict(r)) for r in rows]

   528→    def temporal_window(self, start: str | None, end: str | None,
   529→                        user_id: str | None = None, field: str = "valid",
   530→                        active: bool = True) -> list[Fact]:
   531→        """Zep-compatible temporal queries. field: 'valid' (reality) or 'tx' (recorded).
   532→
   533→        Valid-time uses interval-overlap semantics: a fact matches if its
   534→        [valid_from, valid_to] window intersects [start, end] — so asking
   535→        "where did Alice work in 2025?" retrieves an employment that began
   536→        in 2024 and ended in 2026. Transaction-time uses point semantics.
   537→        """
   538→        clauses, fparams, _ = self._fact_filters(user_id=user_id, active=active)
   539→        cond: list[str] = []
   540→        cparams: list = []
   541→        if field == "valid":
   542→            if start:
   543→                cond.append("(valid_to IS NULL OR valid_to>=?)")
   544→                cparams.append(start[:10])
   545→            if end:
   546→                cond.append("valid_from<=?")
   547→                cparams.append(end[:10])
   548→            order = "valid_from"
   549→        else:
   550→            if start:
   551→                cond.append("tx_from>=?"); cparams.append(start[:10])
   552→            if end:
   553→                cond.append("tx_from<=?"); cparams.append(end[:10])
   554→            order = "tx_from"
   555→        params = cparams + fparams
   556→        where = " AND ".join(cond + clauses) if (cond + clauses) else ""
   557→        q = f"SELECT {FACT_COLUMNS} FROM facts{' WHERE ' + where if where else ''} ORDER BY {order}"
   558→        rows = self.conn.execute(q, params).fetchall()
   559→        return [Fact.from_row(dict(r)) for r in rows]

   561→    def count_facts(self, user_id: str | None = None, active_only: bool = True) -> int:
   562→        clauses, params, _ = self._fact_filters(user_id=user_id, active=active_only)
   563→        where = (" WHERE " + " AND ".join(clauses)) if clauses else ""
   564→        row = self.conn.execute(f"SELECT COUNT(*) c FROM facts{where}", params).fetchone()
   565→        return int(row["c"])

   567→    # ---------------------------------------------------------- active set
   568→    def active_ids(self, branch: str) -> frozenset[str]:
   569→        cached = self._active_cache.get(branch)
   570→        if cached is not None:
   571→            return cached
   572→        head = self.head(branch)
   573→        if head is None:
   574→            return frozenset()
   575→        anc = self.ancestry(head)
   576→        rows = self.conn.execute(
   577→            "SELECT id, birth_commit, retired_commit FROM facts "
   578→            "WHERE birth_commit IS NOT NULL").fetchall()
   579→        ids = frozenset(
   580→            r["id"] for r in rows
   581→            if r["birth_commit"] in anc and (
   582→                not r["retired_commit"] or r["retired_commit"] not in anc))
   583→        if len(self._active_cache) < 16:
   584→            self._active_cache[branch] = ids
   585→        return ids

   587→    def _invalidate(self, branch: str | None = None) -> None:
   588→        if branch:
   589→            self._active_cache.pop(branch, None)
   590→        else:
   591→            self._active_cache.clear()
   592→        self._ancestry_cache.clear()

   601→    def edges_of(self, fact_id: str, kind: str | None = None, direction: str = "out") -> list[dict]:
   602→        outs, ins = [], []
   603→        if direction in ("out", "both"):
   604→            q = "SELECT * FROM edges WHERE src=?" + (" AND kind=?" if kind else "")
   605→            rows = self.conn.execute(q, (fact_id, kind) if kind else (fact_id,)).fetchall()
   606→            outs = [dict(r, dir="out") for r in rows]
   607→        if direction in ("in", "both"):
   608→            q = "SELECT * FROM edges WHERE dst=?" + (" AND kind=?" if kind else "")
   609→            rows = self.conn.execute(q, (fact_id, kind) if kind else (fact_id,)).fetchall()
   610→            ins = [dict(r, dir="in") for r in rows]
   611→        return outs + ins

   613→    def edges_of_many(self, fact_ids: list[str], kind: str | None = None) -> list[dict]:
   614→        """Edges among a set of fact ids (both directions), batched."""
   615→        if not fact_ids:
   616→            return []
   617→        out: list[dict] = []
   618→        B = 200
   619→        for i in range(0, len(fact_ids), B):
   620→            chunk = fact_ids[i:i + B]
   621→            qm = ",".join("?" * len(chunk))
   622→            q = (f"SELECT * FROM edges WHERE src IN ({qm}) AND dst IN ({qm})"
   623→                 + (f" AND kind=?" if kind else ""))
   624→            rows = self.conn.execute(q, chunk + chunk + ([kind] if kind else []))
   625→            out.extend(dict(r) for r in rows)
   626→        return out
