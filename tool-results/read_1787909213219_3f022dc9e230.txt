    53→    53→class TokenBucket:
    54→    54→    def __init__(self, rate: float, burst: int) -> None:
    55→    55→        self.rate = rate
    56→    56→        self.burst = burst
    57→    57→        self._tokens: dict[str, list[float]] = {}
    58→    58→        self._lock = threading.Lock()
    59→    59→
    60→    60→    def allow(self, key: str) -> bool:
    61→    61→        now = time.monotonic()
    62→    62→        with self._lock:
    63→    63→            tokens, ts = self._tokens.get(key, [float(self.burst), now])
    64→    64→            refill = (now - ts) * self.rate
    65→    65→            tokens = min(float(self.burst), tokens + refill)
    66→    66→            if tokens < 1.0:
    67→    67→                self._tokens[key] = [tokens, now]
    68→    68→                return False
    69→    69→            self._tokens[key] = [tokens - 1.0, now]
    70→    70→            return True

   183→   183→def build_handler(state: FabricState):
   184→   184→
   185→   185→    class Handler(BaseHTTPRequestHandler):
   186→   186→        server_version = "context-m/1.0"
   187→   187→        protocol_version = "HTTP/1.1"
   188→   188→
   189→   189→        # -------------------------------------------------------- plumbing
   190→   190→        def log_message(self, fmt, *args):  # quiet access log
   191→   191→            pass
   192→   192→
   193→   193→        def _send(self, code: int, payload, content_type="application/json"):
   194→   194→            body = (payload if isinstance(payload, (bytes, bytearray))
   195→   195→                    else json.dumps(payload, default=str).encode())
   196→   196→            self.send_response(code)
   197→   197→            self.send_header("Content-Type", content_type)
   198→   198→            self.send_header("Content-Length", str(len(body)))
   199→   199→            self.end_headers()
   200→   200→            try:
   201→   201→                self.wfile.write(body)
   202→   202→            except (BrokenPipeError, ConnectionResetError):
   203→   203→                pass
   204→   204→
   205→   205→        def _body(self) -> dict:
   206→   206→            length = int(self.headers.get("Content-Length") or 0)
   207→   207→            if length > MAX_BODY:
   208→   208→                raise ValueError("body too large")
   209→   209→            if not length:
   210→   210→                return {}
   211→   211→            raw = self.rfile.read(length)
   212→   212→            try:
   213→   213→                return json.loads(raw or b"{}")
   214→   214→            except json.JSONDecodeError:
   215→   215→                raise ValueError("invalid JSON body")
   216→   216→
   217→   217→        def _auth(self, action: str) -> dict:
   218→   218→            hdr = self.headers.get("Authorization") or ""
   219→   219→            if not hdr.startswith("Bearer "):
   220→   220→                REGISTRY.inc("contextm_http_requests_total",
   221→   221→                             {"code": "401"})
   222→   222→                self._send(401, {"error": "missing bearer token"})
   223→   223→                return {}
   224→   224→            key = hdr[7:].strip()
   225→   225→            meta = state.keys.verify(key)
   226→   226→            if meta is None:
   227→   227→                state.memory.audit_log.log("auth.failure",
   228→   228→                                       actor=key[:16] + "…",
   229→   229→                                       outcome="invalid_key")
   230→   230→                REGISTRY.inc("contextm_http_requests_total",
   231→   231→                             {"code": "401"})
   232→   232→                self._send(401, {"error": "invalid or revoked key"})
   233→   233→                return {}
   234→   234→            if not state.bucket.allow(key):
   235→   235→                REGISTRY.inc("contextm_http_requests_total",
   236→   236→                             {"code": "429"})
   237→   237→                self._send(429, {"error": "rate limit exceeded"})
   238→   238→                return {}
   239→   239→            try:
   240→   240→                authorize(meta, action)
   241→   241→            except RBACError as e:
   242→   242→                state.memory.audit_log.log(action, actor=meta.get("label")
   243→   243→                                       or meta.get("id", "key"),
   244→   244→                                       role=meta.get("role"),
   245→   245→                                       outcome="denied")
   246→   246→                REGISTRY.inc("contextm_http_requests_total",
   247→   247→                             {"code": "403"})
   248→   248→                self._send(403, {"error": str(e)})
   249→   249→                return {}
   250→   250→            REGISTRY.inc("contextm_http_requests_total", {"code": "200"})
   251→   251→            return meta

   356→   356→        def do_POST(self):
   357→   357→            path = self.path.split("?")[0].rstrip("/")
   358→   358→            t0 = time.monotonic()
   359→   359→            try:
   360→   360→                body = self._body()
   361→   361→            except ValueError as e:
   362→   362→                return self._send(400, {"error": str(e)})
   363→   363→
   364→   364→            routes = {
   365→   365→                "/v1/add": ("memory.add", self._h_add),
   366→   366→                "/v1/search": ("memory.search", self._h_search),
   367→   367→                "/v1/keys": ("keys.create", self._h_key_create),
   368→   368→                "/v1/snapshot": ("governance.snapshot", self._h_snapshot),
   369→   369→                "/v1/restore": ("governance.restore", self._h_restore),
   370→   370→                "/v1/erase": ("governance.erase", self._h_erase),
   371→   371→                "/v1/retention": ("governance.retention", self._h_retention),
   372→   372→                "/v1/state_at": ("governance.pitr", self._h_state_at),
   373→   373→                "/v1/sparql": ("sparql.query", self._h_sparql),
   374→   374→                "/v1/export": ("memory.export", self._h_export),
   375→   375→                "/v1/consolidate": ("governance.consolidate",
   376→   376→                                       self._h_consolidate),
   377→   377→                "/v1/chaos": ("memory.chaos_ingest", self._h_chaos),
   378→   378→            }
   379→   379→            if path not in routes:
   380→   380→                return self._send(404, {"error": f"no route {path}"})
   381→   381→            action, fn = routes[path]
   382→   382→            meta = self._auth(action)
   383→   383→            if not meta:
   384→   384→                return
   385→   385→            actor = meta.get("label") or meta.get("id", "key")
   386→   386→            try:
   387→   387→                out = fn(body, actor, meta)
   388→   388→                REGISTRY.observe("contextm_http_request_seconds",
   389→   389→                                 time.monotonic() - t0)
   390→   390→                return self._send(200, out)
   391→   391→            except RBACError as e:
   392→   392→                return self._send(403, {"error": str(e)})
   393→   393→            except Exception as e:  # noqa: BLE001
   394→   394→                REGISTRY.inc("contextm_http_requests_total",
   395→   395→                             {"code": "500"})
   396→   396→                state.memory.audit_log.log(action, actor=actor,
   397→   397→                                       role=meta.get("role"),
   398→   398→                                       outcome="error",
   399→   399→                                       meta={"error": str(e)[:200]})
   400→   400→                return self._send(500, {"error": str(e)[:500]})

   506→   506→            """Inline SPARQL endpoint — auth'd, shares one Memory.
   507→   507→
   508→   508→            GET  /v1/sparql?query=SELECT...
   509→   509→            POST /v1/sparql  {"query": "SELECT ..."}
   510→   510→            """
   511→   511→            # body is {} for GET — parse query string instead
   512→   512→            query = (body.get("query") if body else None) or ""
   513→   513→            if not query:
   514→   514→                from urllib.parse import parse_qs, urlparse
   515→   515→                q = parse_qs(urlparse(self.path).query)
   516→   516→                query = q.get("query", [""])[0]
   517→   517→            if not query:
   518→   518→                return {"error": "missing 'query' parameter"}
   519→   519→            try:
   520→   520→                from context_m.server.sparql import (
   521→   521→                    execute_sparql, edge_triples)
   522→   522→                with state.lock:
   523→   523→                    fact_objs = state.memory.store.query_facts(
   524→   524→                        active=True,
   525→   525→                        user_id=body.get("user_id") if body else None)
   526→   526→                    facts = [(f.subject, f.relation, f.value)
   527→   527→                             for f in fact_objs]
   528→   528→                    edges = edge_triples(state.memory,
   529→   529→                                          user_id=body.get("user_id")
   530→   530→                                          if body else None)
   531→   531→                    arena = getattr(state.memory, "blob_arena", None)
   532→   532→                    blob_resolver = None
   533→   533→                    if arena is not None:
   534→   534→                        def _resolve(_s, _r, val):
   535→   535→                            from context_m.trace.blob_arena import \
   536→   536→                                get_chunk_text
   537→   537→                            return get_chunk_text(state.memory.store,
   538→   538→                                                   arena, val)
   539→   539→                        blob_resolver = _resolve
   540→   540→                out = execute_sparql(query, facts,
   541→   541→                                       user_id=body.get("user_id")
   542→   542→                                       if body else None,
   543→   543→                                       edge_triples=edges,
   544→   544→                                       blob_resolver=blob_resolver)
   545→   545→                state.memory.audit_log.log(
   546→   546→                    "sparql.query", actor=actor, role=meta.get("role"),
   547→   547→                    meta={"n_results": out.get("n_results", 0),
   548→   548→                           "query_head": query[:80]})
   549→   549→                return out
   550→   550→            except ValueError as e:
   551→   551→                return {"error": str(e)}
   552→   552→            except Exception as e:  # noqa: BLE001
   553→   553→                return {"error": f"server: {e}"}

   555→   555→        def _h_export(self, body, actor, meta):
   556→   556→            """Export facts via the swappable decoder.
   557→   557→
   558→   558→            GET /v1/export?format=rdf|json|datalog|llm_prompt&user_id=X
   559→   559→            """
   560→   560→            from urllib.parse import parse_qs, urlparse
   561→   561→            q = parse_qs(urlparse(self.path).query)
   562→   562→            fmt = (q.get("format") or ["llm_prompt"])[0]
   563→   563→            user_id = (q.get("user_id") or [None])[0]
   564→   564→            from context_m.bridge.decoders import get_decoder
   565→   565→            try:
   566→   566→                decoder = get_decoder(fmt)
   567→   567→            except ValueError as e:
   568→   568→                return {"error": str(e)}
   569→   569→            with state.lock:
   570→   570→                facts = state.memory.store.query_facts(
   571→   571→                    active=True, user_id=user_id)
   572→   572→                # score by recency for export ordering
   573→   573→                scores = {f.id: 1.0 for f in facts}
   574→   574→                out = decoder.render(
   575→   575→                    query="", intent="export", facts=list(facts),
   576→   576→                    scores=scores, notes=None,
   577→   577→                    store=state.memory.store)
   578→   578→            state.memory.audit_log.log(
   579→   579→                "memory.export", actor=actor, role=meta.get("role"),
   580→   580→                resource=user_id or "default",
   581→   581→                meta={"format": fmt, "n_facts": len(facts)})
   582→   582→            # for json decoder, return parsed; otherwise return text
   583→   583→            if fmt == "json":
   584→   584→                try:
   585→   585→                    return json.loads(out)
   586→   586→                except json.JSONDecodeError:
   587→   587→                    pass
   588→   588→            return {"format": fmt, "n_facts": len(facts),
   589→   589→                    "content": out}

   638→   638→# ------------------------------------------------------------------ launch
   639→   639→def serve(memory: Memory | None = None, host: str = "0.0.0.0",
   640→   640→          port: int = 8900, *, sparql_port: int | None = None,
   641→   641→          sparql_host: str = "0.0.0.0",
   642→   642→          sparql_user_id: str | None = None) -> ThreadingHTTPServer:
   643→   643→    if memory is None:
   644→   644→        memory = Memory(Config.from_env())
   645→   645→    state = FabricState(memory)
   646→   646→    httpd = ThreadingHTTPServer((host, port), build_handler(state))
   647→   647→    httpd.daemon_threads = True
   648→   648→    # optionally co-host a SPARQL endpoint sharing the same Memory instance
   649→   649→    if sparql_port:
   650→   650→        from context_m.server.sparql import SparqlServer
   651→   651→        sparql = SparqlServer(memory, host=sparql_host, port=sparql_port,
   652→   652→                               user_id=sparql_user_id)
   653→   653→        sparql.start_background()
   654→   654→        httpd.sparql = sparql  # type: ignore[attr-defined]
   655→   655→    return httpd

   658→   658→def main() -> None:
   659→   659→    ap = argparse.ArgumentParser(prog="contextm-serve",
   660→   660→                                 description="Context-M REST API server")
   661→   661→    ap.add_argument("--db", default=":memory:", help="SQLite path")
   662→   662→    ap.add_argument("--host", default="0.0.0.0")
   663→   663→    ap.add_argument("--port", type=int, default=8900)
   664→   664→    ap.add_argument("--pii", default=None,
   665→   665→                    help="off|redact|block|tag (default: config/env)")
   666→   666→    ap.add_argument("--admin-key", default=None,
   667→   667→                    help="create this admin API key at boot (dev convenience)")
   668→   668→    ap.add_argument("--sparql-port", type=int, default=None,
   669→   669→                    help="co-host a SPARQL endpoint on this port "
   670→   670→                         "(shares one Memory instance with the REST API)")
   671→   671→    ap.add_argument("--sparql-host", default="0.0.0.0",
   672→   672→                    help="bind SPARQL endpoint to this host")
   673→   673→    ap.add_argument("--sparql-user-id", default=None,
   674→   674→                    help="scope SPARQL queries to a single user")
   675→   675→    args = ap.parse_args()
   676→   676→
   677→   677→    cfg = Config.from_env(db_path=args.db)
   678→   678→    if args.pii:
   679→   679→        cfg.pii_mode = args.pii
   680→   680→    memory = Memory(cfg)
   681→   681→    if args.admin_key:
   682→   682→        meta = memory.keys.create("admin", label="boot-admin")
   683→   683→        print(f"[context-m] admin API key (save now, shown once): {meta['key']}")
   684→   684→    httpd = serve(memory, args.host, args.port,
   685→   685→                  sparql_port=args.sparql_port,
   686→   686→                  sparql_host=args.sparql_host,
   687→   687→                  sparql_user_id=args.sparql_user_id)
   688→   688→    print(f"[context-m] REST API on http://{args.host}:{args.port}"
   689→   689→          f"  (db={args.db}, pii={cfg.pii_mode}, "
   690→   690→          f"encrypt={cfg.encryption_at_rest})")
   691→   691→    if args.sparql_port:
   692→   692→        print(f"[context-m] SPARQL endpoint on http://{args.sparql_host}:"
   693→   693→              f"{args.sparql_port}/  (co-hosted, shares Memory)")
   694→   694→
   695→   695→    # graceful shutdown on SIGTERM/SIGINT — drain in-flight requests
   696→   696→    def _shutdown(signum, frame):
   697→   697→        print(f"\n[context-m] received signal {signum} — shutting down...")
   698→   698→        sparql = getattr(httpd, "sparql", None)
   699→   699→        if sparql is not None:
   700→   700→            sparql.stop()
   701→   701→        httpd.shutdown()  # unblocks serve_forever()
   702→   702→    signal.signal(signal.SIGTERM, _shutdown)
   703→   703→    signal.signal(signal.SIGINT, _shutdown)
   704→   704→    try:
   705→   705→        httpd.serve_forever()
   706→   706→    except KeyboardInterrupt:
   707→   707→        pass
   708→   708→    finally:
   709→   709→        memory.close()
   710→   710→
   711→   711→
   712→   712→if __name__ == "__main__":
   713→   713→    main()
