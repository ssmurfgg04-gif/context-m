   255→class MCPServer:
   256→    def __init__(self, memory: Memory) -> None:
   257→        self.memory = memory
   258→
   259→    # ------------------------------------------------------------------
   260→    def handle(self, request: dict) -> dict | None:
   261→        method = request.get("method", "")
   262→        req_id = request.get("id")
   263→        if method == "initialize":
   264→            return self._ok(req_id, {
   265→                "protocolVersion": PROTOCOL_VERSION,
   266→                "capabilities": {"tools": {"listChanged": False}},
   267→                "serverInfo": SERVER_INFO,
   268→            })
   269→        if method == "notifications/initialized":
   270→            return None
   271→        if method == "ping":
   272→            return self._ok(req_id, {})
   273→        if method == "tools/list":
   274→            return self._ok(req_id, {"tools": TOOLS})
   275→        if method == "tools/call":
   276→            return self._ok(req_id, self._call_tool(
   277→                request.get("params", {}).get("name", ""),
   278→                request.get("params", {}).get("arguments", {}) or {}))
   279→        if method == "resources/list":
   280→            return self._ok(req_id, {"resources": []})
   281→        if req_id is not None:
   282→            return self._err(req_id, -32601, f"method not found: {method}")
   283→        return None
   284→

   287→        m = self.memory
   288→        try:

   290→                out = m.add(args.get("messages"),
   291→                            user_id=args.get("user_id", "default"),

   293→                            run_id=args.get("run_id"),
   294→                            timestamp=args.get("timestamp"))
   295→                text = json.dumps({"stored": len(out.get("results", [])),
   296→                                   "stats": out.get("stats")}, default=str)
   297→            elif name == "contextm_search":
   298→                out = m.search(args.get("query", ""),
   299→                               user_id=args.get("user_id", "default"),
   300→                               limit=args.get("limit", 12))

   303→                out = m.get_all(user_id=args.get("user_id", "default"),

   400→    # ------------------------------------------------------------------
   401→    def _query_extract(self, query: str, user_id: str = "default",
   402→                       k: int = 5) -> dict:
   403→        """Hybrid query-time extraction. Uses QueryTimeExtractor if
   404→        available; falls back to standard search results if not."""
   405→        try:
   406→            from context_m.bridge.query_extract import QueryTimeExtractor
   407→            from context_m.text.dissim import DisSimSplitter
   408→            from context_m.text.idiolect import PerUserIdiolectNormalizer
   409→            from context_m.text.embedder import HashingEmbedder
   410→
   411→            palace = self.memory.palace
   412→            store = self.memory.store
   413→            embedder = HashingEmbedder(palace.dims, palace.cfg.seed)
   414→            dissim = DisSimSplitter(max_depth=2)
   415→            idiolect = PerUserIdiolectNormalizer(embedder)
   416→            extractor = QueryTimeExtractor(
   417→                palace, store, embedder, dissim=dissim,
   418→                idiolect=idiolect,
   419→                pattern_extractor=self.memory.extractor if hasattr(
   420→                    self.memory, "extractor") else None)
   421→            results = extractor.query(query, user_id=user_id, k=k)
   422→            return {
   423→                "query": query,
   424→                "user_id": user_id,
   425→                "extracted_count": len(results),
   426→                "results": results,
   427→                "path": "query_time_pattern",
   428→            }
   429→        except Exception as e:
   430→            # graceful fallback: standard search
   431→            out = self.memory.search(query, user_id=user_id, limit=k)
   432→            return {
   433→                "query": query,
   434→                "user_id": user_id,
   435→                "fallback": "standard_search",
   436→                "error": str(e),
   437→                "context_block": out.get("context_block", ""),
   438→            }

   440→    def _attribution(self, query: str, user_id: str = "default") -> dict:
   441→        """ProtoDash attribution for a query — which source chunks contributed."""
   442→        try:
   443→            from context_m.vsa.attribution import ProtoDashAttributer, sentence_level_score
   444→            from context_m.text.embedder import HashingEmbedder
   445→            import numpy as np
   446→
   447→            # standard search to get candidates
   448→            out = self.memory.search(query, user_id=user_id, limit=10)
   449→            results = out.get("results", [])
   450→            if not results:
   451→                return {"query": query, "attributions": []}
   452→            embedder = HashingEmbedder(self.memory.palace.dims,
   453→                                       self.memory.palace.cfg.seed)
   454→            q_emb = embedder.embed(query)
   455→            cand_embs = np.stack([embedder.embed(r.get("memory", ""))
   456→                                   for r in results])
   457→            cand_ids = [r.get("id", str(i)) for i, r in enumerate(results)]
   458→            attrib = ProtoDashAttributer(kernel="linear")
   459→            weights = attrib.attribute(q_emb, cand_embs, cand_ids, m=5)
   460→            return {
   461→                "query": query,
   462→                "user_id": user_id,
   463→                "attributions": [
   464→                    {"fact_id": fid, "weight": w, "memory": r.get("memory", "")}
   465→                    for (fid, w), r in zip(weights, results)
   466→                ],
   467→            }
   468→        except Exception as e:
   469→            return {"query": query, "error": str(e), "attributions": []}

   471→    def _zk_prove(self, memory_id: str, public_commitment: str,
   472→                  threshold: int = 32) -> dict:
   473→        """Hamming ZK proof on binary codec vectors."""
   474→        try:
   475→            from context_m.security.zk_hamming import HammingZKProver
   476→            palace = self.memory.palace
   477→            # fetch the memory's packed vector
   478→            row = palace._id2row.get(memory_id)
   479→            if row is None:
   480→                return {"error": f"memory {memory_id} not in palace"}
   481→            packed = palace._packed[row]
   482→            # convert public commitment hex → bytes
   483→            public = bytes.fromhex(public_commitment)
   484→            private = bytes(packed.tobytes())
   485→            prover = HammingZKProver(dims=palace.dims, threshold=threshold)
   486→            proof = prover.prove(public, private)
   487→            verified = prover.verify(public, proof)
   488→            return {
   489→                "memory_id": memory_id,
   490→                "verified": verified,
   491→                "weight": proof.weight,
   492→                "threshold": proof.threshold,
   493→                "commitment": proof.commitment[:32] + "...",
   494→            }
   495→        except Exception as e:
   496→            return {"memory_id": memory_id, "error": str(e)}

   498→    # ------------------------------------------------------------------
   499→    def _reconstruct(self, query: str, user_id: str = "default",
   500→                     k: int = 10, max_hops: int = 3) -> dict:
   501→        """MRAgent-style active memory reconstruction.
   502→
   503→        Uses Memory.reader.reconstruct() (which exists from prior work
   504→        — bridge/reader.py). Falls back to standard search if the
   505→        reconstruct path is disabled in Config.
   506→        """
   507→        try:
   508→            reader = getattr(self.memory, "reader", None)
   509→            if reader is None:
   510→                out = self.memory.search(query, user_id=user_id, limit=k)
   511→                return {"query": query, "fallback": "standard_search",
   512→                        "context_block": out.get("context_block", "")}
   513→            reconstruct_fn = getattr(reader, "reconstruct", None)
   514→            if (reconstruct_fn is None
   515→                    or not getattr(self.memory.config,
   516→                                    "reconstruct_enabled", False)):
   517→                out = self.memory.search(query, user_id=user_id, limit=k * 2)
   518→                return {"query": query, "fallback": "standard_search_wide",
   519→                        "context_block": out.get("context_block", ""),
   520→                        "results": out.get("results", [])[:k]}
   521→            res = reconstruct_fn(query, user_id=user_id, k=k,
   522→                                  max_hops=max_hops)
   523→            return {
   524→                "query": query,
   525→                "user_id": user_id,
   526→                "intent": getattr(res, "intent", "reconstruct"),
   527→                "narrative": getattr(res, "context_block", ""),
   528→                "facts": [
   529→                    {"id": getattr(f, "id", ""), "subject": f.subject,
   530→                     "relation": f.relation, "value": f.value,
   531→                     "confidence": getattr(f, "confidence", 0.0)}
   532→                    for f in getattr(res, "facts", [])
   533→                ],
   534→                "provenance": getattr(res, "provenance", {}),
   535→                "timing": getattr(res, "timing", {}),
   536→            }
   537→        except Exception as e:
   538→            return {"query": query, "error": str(e)}

   540→    # ------------------------------------------------------------------
   541→    def _consolidate(self, user_id: str = "default", dry_run: bool = False,
   542→                     lifecycle: bool = True, dreaming: bool = True,
   543→                     fade: bool = True, tmt: bool = True) -> dict:
   544→        """On-demand consolidation: FadeMem + TMT + Aeon dreaming.
   545→
   546→        Wraps Memory.consolidate() with explicit fade/tmt toggles so
   547→        callers can run a partial pass (e.g. fade-only) on demand.
   548→        Idempotent — safe to call repeatedly.
   549→        """
   550→        try:
   551→            out = self.memory.consolidate(
   552→                lifecycle=lifecycle, dreaming=dreaming,
   553→                run_fade=fade, run_tmt=tmt, user_id=user_id,
   554→                dry_run=dry_run)
   555→            return {"user_id": user_id, "dry_run": dry_run,
   556→                    "lifecycle": out.get("lifecycle", {}),
   557→                    "dreaming": out.get("dreaming", {})}
   558→        except Exception as e:
   559→            return {"user_id": user_id, "error": str(e)}

   561→    # ------------------------------------------------------------------
   562→    def _working_memory(self, query: str, user_id: str = "default",
   563→                        k: int = 12) -> dict:
   564→        """Compress top-k retrieved facts into a single HRR superposition."""
   565→        try:
   566→            reader = getattr(self.memory, "reader", None)
   567→            if reader is None:
   568→                return {"query": query, "error": "reader not available"}
   569→            return reader.working_memory(query, user_id=user_id, k=k)
   570→        except Exception as e:
   571→            return {"query": query, "error": str(e)}

   574→    def _hologram_extract(self, hrr_b64: str, role: str,
   575→                          candidate_ids: list[str],
   576→                          user_id: str = "default") -> dict:

   600→def serve(db_path: str | None = None) -> None:
   601→    cfg = Config.from_env()
   602→    if db_path:
   603→        cfg.db_path = db_path
   604→    memory = Memory(cfg)
   605→    server = MCPServer(memory)
   606→    for line in sys.stdin:
   607→        line = line.strip()
   608→        if not line:
   609→            continue
   610→        try:
   611→            request = json.loads(line)
   612→        except json.JSONDecodeError:
   613→            continue
   614→        response = server.handle(request)
   615→        if response is not None:
   616→            sys.stdout.write(json.dumps(response) + "\n")
   617→            sys.stdout.flush()
   618→
   619→
   620→if __name__ == "__main__":
   621→    serve()
