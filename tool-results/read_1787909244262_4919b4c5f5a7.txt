     1→"""SPARQL endpoint — non-LLM decoder path (NSR-inspired, v2 extended).
     2→
     3→arXiv insight (NSR / ESWEEK24): the VSA core is task-agnostic; only
     4→the decoder changes. Context-M's reader was hardcoded to format facts
     5→for an LLM prompt. The Decoders module extracts that formatter into
     6→a pluggable interface — `RDFDecoder` exports facts as RDF/N3 triples.
     7→
     8→This module exposes a SPARQL HTTP endpoint that:
     9→  1. accepts SPARQL SELECT queries via HTTP GET/POST
    10→  2. parses the query (a hand-rolled parser covering the common
    11→     subset: SELECT [DISTINCT] ?vars WHERE { triple-patterns +
    12→     FILTERs + OPTIONAL } ORDER BY ?var [ASC|DESC] LIMIT N)
    13→  3. retrieves facts from Context-M's Memory.store (or its
    14→     RDFDecoder if attached — same substrate either way)
    15→  4. runs the WHERE clause as a join pipeline over those triples
    16→     — naive nested-loop join with binding propagation, OPTIONAL
    17→     handled via LEFT-JOIN semantics, FILTER applied per binding
    18→  5. resolves blob-arena-stored objects on demand (the sidecar
    19→     blob arena is the Aeon off-graph store; SPARQL "o" may
    20→     transparently dereference long text via Memory.get_chunk_text)
    21→  6. exposes CAUSAL / REFERS_TO typed edges (Aeon) via the
    22→     `edge/2` predicate family so external graph tools can walk
    23→     the truth-maintenance and episodic-atlas graphs natively
    24→  7. returns the result as SPARQL 1.1 JSON Results
    25→
    26→This is a NON-LLM retrieval path. Zero LLM calls. The same palace +
    27→Trace substrate that powers LLM context-stuffing now serves SPARQL.
    28→
    29→Usage:
    30→    # standalone SPARQL endpoint (e.g. on port 8910)
    31→    python -m context_m.server.sparql --port 8910
    32→
    33→    # launched alongside the REST API (recommended for production):
    34→    # `cortexm serve-rest --sparql-port 8910` — both share one Memory
    35→    # instance; the REST API also exposes /v1/sparql for unified access.
    36→
    37→Example queries (v2 supports a much richer subset than v1):
    38→    SELECT ?s ?p ?o WHERE { ?s ?p ?o } LIMIT 10
    39→    SELECT DISTINCT ?s WHERE { ?s ?p ?o } ORDER BY ?s
    40→    SELECT ?s ?o WHERE { ?s "name" ?o . FILTER regex(?o, "^Jen", "i") }
    41→    SELECT ?cause ?effect WHERE {
    42→        ?cause edge:CAUSAL ?effect .
    43→        ?effect "name" "Alice"
    44→    } LIMIT 5
    45→
    46→LIMITATIONS (honest, documented):
    47→  - Parser covers the SELECT subset named above. UNION, CONSTRUCT,
    48→    ASK, DESCRIBE, property paths (rdf:type/rdfs:subClassOf*), and
    49→    SPARQL 1.1 aggregates (GROUP BY / COUNT / SUM) are NOT yet
    50→    supported. For those, export via RDFDecoder + use Apache Jena.
    51→  - No inference / reasoning over RDFS/OWL — pattern matching on
    52→    the actual stored facts only.
    53→  - Joins are nested-loop (no query planner). Adequate for the
    54→    ~10^5 facts Context-M is designed to hold per user; for larger
    55→    stores, use a real triple store fed via the RDFDecoder export.
    56→"""

    81→# ----------------------------------------------------------- parser
    82→def parse_sparql(query: str) -> dict:
    83→    """Parse a SPARQL SELECT query.
    84→
    85→    Supports (v2, extended):
    86→        SELECT [DISTINCT] ?var+ WHERE {
    87→            triple-pattern ( ?s ?p ?o . | ?s "literal" ?o . | ... )
    88→            FILTER ( regex(?v, "pat", "i") | ?v = "lit" | ?v != "lit" )
    89→            OPTIONAL { ... }      # left-join semantics
    90→        }
    91→        [ORDER BY ?var [ASC|DESC]]
    92→        [LIMIT N]
    93→        [OFFSET N]
    94→
    95→    Typed-edge shortcut: `edge:KIND` is recognized as a predicate
    96→    where KIND is one of the canonical edge kinds (CAUSAL, REFERS_TO,
    97→    SUPERSEDES, CONTRADICTS, MERGED_WITH, EXTRACTED_FROM,
    98→    TEMPORALLY_PRECEDED_BY, NEXT). The triple (?a edge:CAUSAL ?b)
    99→    queries the Trace's edges table instead of the facts table.
   100→
   101→    Returns:
   102→        {
   103→          "distinct": bool,
   104→          "select":   ["?s", "?p", "?o"],
   105→          "patterns": [("?s", "?p", "?o"), ...],
   106→          "optionals":[[pat, ...], ...],     # one list per OPTIONAL block
   107→          "filters":  [{"type": "regex"/"equals"/"ne", ...}, ...],
   108→          "order_by": {"var": "?s", "desc": False} | None,
   109→          "limit":    int | None,
   110→          "offset":   int | None,
   111→        }
   112→
   113→    Raises ValueError on unparseable input.
   114→    """
   115→    query = query.strip()
   116→    # strip trailing semicolons / whitespace
   117→    while query.endswith(";"):
   118→        query = query[:-1].strip()
   119→
   120→    # match: SELECT [DISTINCT] <select-vars> WHERE { <where> }
   121→    # select-vars may be one or more ?var tokens, or *
   122→    sel_grp = ""
   123→    where_clause = ""
   124→    tail = ""
   125→    distinct_grp = ""
   126→    matched = False
   127→    for pat in (
   128→        # canonical form with WHERE keyword
   129→        r"SELECT\s+(DISTINCT\s+)?((?:\?\S+(?:\s+|$))+|\*\s*?)\s*WHERE\s*\{(.*)\}\s*(.*)$",
   130→        # without WHERE keyword (some demos)
   131→        r"SELECT\s+(DISTINCT\s+)?((?:\?\S+(?:\s+|$))+|\*\s*?)\s*\{(.*)\}\s*(.*)$",
   132→    ):
   133→        m = re.match(pat, query, re.IGNORECASE | re.DOTALL)
   134→        if m:
   135→            distinct_grp = m.group(1) or ""
   136→            sel_grp = m.group(2) or ""
   137→            where_clause = m.group(3) or ""
   138→            tail = m.group(4) or ""
   139→            matched = True
   140→            break
   141→    if not matched:
   142→        raise ValueError(
   143→            "query must be 'SELECT [DISTINCT] ?vars WHERE { ... }' "
   144→            "— more complex SPARQL not yet supported in v2")
   145→
   146→    # parse SELECT vars (may be multiple ?vars separated by spaces, or *)
   147→    is_distinct = bool(distinct_grp)
   148→    if sel_grp.strip() == "*":
   149→        select_vars: list[str] = ["?s", "?p", "?o"]
   150→    else:
   151→        # find all ?vars in the select clause
   152→        select_vars = re.findall(r"\?\w+", sel_grp)
   153→        if not select_vars:
   154→            raise ValueError("SELECT must list at least one variable")
   155→
   156→    # parse tail (ORDER BY / LIMIT / OFFSET — order-insensitive)
   157→    order_by, limit, offset = _parse_tail(tail + " " + where_clause[-0:])
   158→    # NOTE: the WHERE clause may also have a trailing LIMIT etc — we
   159→    # extract the tail BEFORE the where clause's trailing }. The above
   160→    # regex captures tail as everything AFTER the closing }. Good.
   161→
   162→    # parse WHERE: triple patterns, FILTER, OPTIONAL
   163→    where_clause = where_clause.strip()
   164→    patterns: list[tuple[str, str, str]] = []
   165→    filters: list[dict] = []
   166→    optionals: list[list[tuple[str, str, str]]] = []
   167→
   168→    # tokenize while respecting OPTIONAL { ... } nesting, FILTER(...),
   169→    # and quoted strings. Then merge 'FILTER' + the following
   170→    # parenthesized token so _parse_filter sees the full clause.
   171→    tokens = _merge_filter_tokens(_tokenize_where(where_clause))
   172→
   173→    cur_optional: list[tuple[str, str, str]] | None = None
   174→    i = 0
   175→    while i < len(tokens):
   176→        tok = tokens[i].strip()
   177→        if not tok:
   178→            i += 1
   179→            continue
   180→        upper = tok.upper()
   181→        if upper.startswith("OPTIONAL"):
   182→            # expect '{' next
   183→            i += 1
   184→            if i >= len(tokens) or tokens[i].strip() != "{":
   185→                raise ValueError("OPTIONAL must be followed by '{'")
   186→            # gather until matching '}'
   187→            depth = 1
   188→            inner: list[str] = []
   189→            i += 1
   190→            while i < len(tokens) and depth > 0:
   191→                t = tokens[i].strip()
   192→                if t == "{":
   193→                    depth += 1
   194→                    inner.append(t)
   195→                elif t == "}":
   196→                    depth -= 1
   197→                    if depth > 0:
   198→                        inner.append(t)
   199→                else:
   200→                    inner.append(t)
   201→                i += 1
   202→            # parse inner as a sub-clause (patterns + filters)
   203→            sub_pats, sub_filts = _parse_triples(inner)
   204→            # v2 stores optional patterns separately; filters inside
   205→            # OPTIONAL are also applied within the left-join
   206→            optionals.append(sub_pats)
   207→            filters.extend(sub_filts)  # FILTERs are hoisted — applied later
   208→            cur_optional = None
   209→            continue
   210→        if upper.startswith("FILTER"):
   211→            filt = _parse_filter(tok)
   212→            if filt:
   213→                filters.append(filt)
   214→            i += 1
   215→            continue
   216→        # else: triple-pattern token. Look at groups of 3.
   217→        # Easier: re-tokenize by '.' but keep quoted strings together
   218→        # _tokenize_where already split on whitespace, so triple tokens
   219→        # come as a stream. Group until we have 3 non-'.' tokens.
   220→        # Skip '.' separator.
   221→        if tok == ".":
   222→            i += 1
   223→            continue
   224→        # collect 3 terms for a triple
   225→        terms = [tok]
   226→        j = i + 1
   227→        while j < len(tokens) and len(terms) < 3:
   228→            t = tokens[j].strip()
   229→            if t == ".":
   230→                j += 1
   231→                continue
   232→            if t.upper().startswith("FILTER") or t.upper().startswith(
   233→                    "OPTIONAL") or t == "}":
   234→                break
   235→            terms.append(t)
   236→            j += 1
   237→        if len(terms) == 3:
   238→            patterns.append(tuple(terms))
   239→        i = j
   240→
   241→    return {
   242→        "distinct": is_distinct,
   243→        "select": select_vars,
   244→        "patterns": patterns,
   245→        "optionals": optionals,
   246→        "filters": filters,
   247→        "order_by": order_by,
   248→        "limit": limit,
   249→        "offset": offset,
   250→    }
   251→
   252→
   253→def _parse_tail(tail: str) -> tuple[dict | None, int | None, int | None]:
   254→    """Extract ORDER BY / LIMIT / OFFSET from the post-WHERE tail."""
   255→    order_by = None
   256→    limit = None
   257→    offset = None
   258→
   259→    # ORDER BY ?var [ASC|DESC]
   260→    m = re.search(
   261→        r"ORDER\s+BY\s+(\?\w+)(?:\s+(ASC|DESC))?",
   262→        tail, re.IGNORECASE)
   263→    if m:
   264→        order_by = {"var": m.group(1),
   265→                    "desc": (m.group(2) or "").upper() == "DESC"}
   266→
   267→    # LIMIT N
   268→    m = re.search(r"LIMIT\s+(\d+)", tail, re.IGNORECASE)
   269→    if m:
   270→        limit = int(m.group(1))
   271→
   272→    # OFFSET N
   273→    m = re.search(r"OFFSET\s+(\d+)", tail, re.IGNORECASE)
   274→    if m:
   275→        offset = int(m.group(1))
   276→
   277→    return order_by, limit, offset

   280→def _tokenize_where(s: str) -> list[str]:
   281→    """Split WHERE body into tokens, preserving quoted strings + parens."""
   282→    tokens: list[str] = []
   283→    cur = ""
   284→    in_string = False
   285→    quote_char = None
   286→    depth = 0
   287→    for ch in s:
   288→        if in_string:
   289→            cur += ch
   290→            if ch == quote_char:
   291→                in_string = False
   292→            continue
   293→        if ch in ('"', "'"):
   294→            in_string = True
   295→            quote_char = ch
   296→            cur += ch
   297→            continue
   298→        if ch == "(":
   299→            depth += 1
   300→            cur += ch
   301→            continue
   302→        if ch == ")":
   303→            depth -= 1
   304→            cur += ch
   305→            continue
   306→        if ch == "{":
   307→            if cur.strip():
   308→                tokens.append(cur.strip())
   309→                cur = ""
   310→            tokens.append("{")
   311→            continue
   312→        if ch == "}":
   313→            if cur.strip():
   314→                tokens.append(cur.strip())
   315→                cur = ""
   316→            tokens.append("}")
   317→            continue
   318→        if depth > 0:
   319→            cur += ch
   320→            continue
   321→        if ch.isspace():
   322→            if cur.strip():
   323→                tokens.append(cur.strip())
   324→                cur = ""
   325→            continue
   326→        cur += ch
   327→    if cur.strip():
   328→        tokens.append(cur.strip())
   329→    return tokens

   332→def _parse_triples(tokens: list[str]) -> tuple[list[tuple], list[dict]]:
   333→    """Parse a flat list of triple tokens + FILTERs into (pats, filts)."""
   334→    tokens = _merge_filter_tokens(tokens)
   335→    pats: list[tuple] = []
   336→    filts: list[dict] = []
   337→    i = 0
   338→    while i < len(tokens):
   339→        tok = tokens[i].strip()
   340→        if not tok:
   341→            i += 1
   342→            continue
   343→        if tok.upper().startswith("FILTER"):
   344→            f = _parse_filter(tok)
   345→            if f:
   346→                filts.append(f)
   347→            i += 1
   348→            continue
   349→        if tok == ".":
   350→            i += 1
   351→            continue
   352→        terms = [tok]
   353→        j = i + 1
   354→        while j < len(tokens) and len(terms) < 3:
   355→            t = tokens[j].strip()
   356→            if t == ".":
   357→                j += 1
   358→                continue
   359→            if t.upper().startswith("FILTER") or t == "}":
   360→                break
   361→            terms.append(t)
   362→            j += 1
   363→        if len(terms) == 3:
   364→            pats.append(tuple(terms))
   365→        i = j
   366→    return pats, filts

   369→def _parse_filter(tok: str) -> dict | None:
   370→    """Parse one FILTER(...) clause into a filter dict.
   371→
   372→    `tok` may be:
   373→      - 'FILTER regex(?v, "pat", "flags")'  (merged by tokenizer)
   374→      - 'FILTER(?v = "value")'
   375→      - 'FILTER(?v != "value")'
   376→      - 'FILTER' alone (in which case caller should merge next token)
   377→    """
   378→    # FILTER regex(?v, "pat", "flags")
   379→    m = re.match(
   380→        r"FILTER\s+regex\s*\(\s*(\?\w+)\s*,\s*[\"'](.+?)[\"']\s*"
   381→        r"(?:,\s*[\"'](\w+)[\"']\s*)?\)",
   382→        tok, re.IGNORECASE | re.DOTALL)
   383→    if m:
   384→        return {"type": "regex", "var": m.group(1),
   385→                "pattern": m.group(2), "flags": m.group(3) or ""}
   386→    # FILTER regex without space: FILTERregex(...) — unlikely but tolerated
   387→    m = re.match(
   388→        r"FILTERregex\s*\(\s*(\?\w+)\s*,\s*[\"'](.+?)[\"']\s*"
   389→        r"(?:,\s*[\"'](\w+)[\"']\s*)?\)",
   390→        tok, re.IGNORECASE | re.DOTALL)
   391→    if m:
   392→        return {"type": "regex", "var": m.group(1),
   393→                "pattern": m.group(2), "flags": m.group(3) or ""}
   394→    # FILTER(?v = "value")
   395→    m = re.match(
   396→        r"FILTER\s*\(\s*(\?\w+)\s*=\s*[\"'](.+?)[\"']\s*\)",
   397→        tok, re.IGNORECASE | re.DOTALL)
   398→    if m:
   399→        return {"type": "equals", "var": m.group(1), "value": m.group(2)}
   400→    # FILTER(?v != "value")
   401→    m = re.match(
   402→        r"FILTER\s*\(\s*(\?\w+)\s*!=\s*[\"'](.+?)[\"']\s*\)",
   403→        tok, re.IGNORECASE | re.DOTALL)
   404→    if m:
   405→        return {"type": "ne", "var": m.group(1), "value": m.group(2)}
   406→    return None
   407→
   408→
   409→def _merge_filter_tokens(tokens: list[str]) -> list[str]:
   410→    """If 'FILTER' appears as a standalone token, merge it with the
   411→    following parenthesized token so _parse_filter sees the full
   412→    'FILTER regex(...)' or 'FILTER(...)' string.
   413→    """
   414→    out: list[str] = []
   415→    i = 0
   416→    while i < len(tokens):
   417→        t = tokens[i]
   418→        if t.upper() == "FILTER" and i + 1 < len(tokens):
   419→            merged = t + " " + tokens[i + 1]
   420→            out.append(merged)
   421→            i += 2
   422→            continue
   423→        out.append(t)
   424→        i += 1
   425→    return out

   428→# ----------------------------------------------------------- matcher
   429→def match_triple(triple: tuple, pattern: tuple,
   430→                 bindings: dict | None = None) -> dict | None:
   431→    """Try to match a triple against a pattern, returning extended
   432→    bindings or None if no match.
   433→
   434→    triple:   (subject, relation, value) from Context-M facts
   435→    pattern:  ("?s", "?p", "?o") or with literals like
   436→              ("?s", "name", "?o")
   437→    """
   438→    if len(triple) != 3 or len(pattern) != 3:
   439→        return None
   440→    b = dict(bindings or {})
   441→    for tri_elem, pat_elem in zip(triple, pattern):
   442→        if pat_elem.startswith("?"):
   443→            if pat_elem in b:
   444→                if b[pat_elem] != tri_elem:
   445→                    return None
   446→            else:
   447→                b[pat_elem] = tri_elem
   448→        else:
   449→            # literal: strip quotes
   450→            literal = pat_elem.strip('"\'')
   451→            if literal != tri_elem:
   452→                return None
   453→    return b
   454→

   456→def apply_filters(bindings: dict, filters: list[dict]) -> bool:
   457→    """Return True if `bindings` satisfies all `filters`."""
   458→    for f in filters:
   459→        var = f["var"]
   460→        if var not in bindings:
   461→            return False
   462→        val = bindings[var]
   463→        if f["type"] == "regex":
   464→            flags = 0
   465→            if "i" in f.get("flags", ""):
   466→                flags |= re.IGNORECASE
   467→            if not re.search(f["pattern"], val, flags):
   468→                return False
   469→        elif f["type"] == "equals":
   470→            if val != f["value"]:
   471→                return False
   472→        elif f["type"] == "ne":
   473→            if val == f["value"]:
   474→                return False
   475→    return True

   482→_EDGE_PRED_TO_KIND = {
   483→    "edge:CAUSAL": CAUSAL,
   484→    "edge:REFERS_TO": REFERS_TO,
   485→    "edge:SUPERSEDES": SUPERSEDS,
   486→    "edge:CONTRADICTS": CONTRADICTS,

   488→    "edge:EXTRACTED_FROM": EXTRACTED_FROM,
   489→    "edge:TEMPORALLY_PRECEDED_BY": TEMPORALLY_PRECEDED_BY,
   490→    "edge:NEXT": NEXT,
   491→}

   494→def is_edge_predicate(p: str) -> bool:
   495→    """Is this a typed-edge predicate (edge:KIND)?"""
   496→    return p in _EDGE_PRED_TO_KIND

   499→def edge_triples(mem, user_id: str | None = None) -> list[tuple]:
   500→    """Yield all typed edges as (src, edge:KIND, dst) triples.
   501→
   502→    Lets SPARQL queries like `?a edge:CAUSAL ?b` walk the typed-edge
   503→    graph natively. We materialize on demand — the edges table is
   504→    usually 10^4-10^5 rows so this is fast.
   505→    """
   506→    rows = mem.store.conn.execute(
   507→        "SELECT src, kind, dst FROM edges").fetchall()
   508→    out = []
   509→    for src, kind, dst in rows:
   510→        # reverse-map kind to SPARQL-facing predicate
   511→        for sp_pred, k in _EDGE_PRED_TO_KIND.items():
   512→            if k == kind:
   513→                out.append((src, sp_pred, dst))
   514→                break
   515→    return out

   518→# ----------------------------------------------------------- executor
   519→def execute_sparql(query: str, facts: list,
   520→                   user_id: str | None = None,
   521→                   *, edge_triples: list | None = None,
   522→                   blob_resolver=None) -> dict:
   523→    """Execute a parsed SPARQL query against a list of facts.
   524→
   525→    facts: list of (subject, relation, value) triples from the facts table
   526→    edge_triples: optional list of (src, edge:KIND, dst) triples from the
   527→                  edges table (for typed-edge queries). If None, edge:
   528→                  predicates won't match anything.
   529→    blob_resolver: optional callable(subject, relation, value) -> str that
   530→                  resolves long object values stored in the sidecar blob
   531→                  arena. If a SPARQL FILTER references the object text
   532→                  and the value is a chunk_id pointing into the arena,
   533→                  the resolver dereferences on demand.
   534→
   535→    Returns SPARQL JSON Results format dict.
   536→    """
   537→    parsed = parse_sparql(query)
   538→    select_vars = parsed["select"]
   539→    patterns = parsed["patterns"]
   540→    optionals = parsed["optionals"]
   541→    filters = parsed["filters"]
   542→    order_by = parsed["order_by"]
   543→    limit = parsed["limit"]
   544→    offset = parsed["offset"] or 0
   545→
   546→    if not patterns and not optionals:
   547→        # empty WHERE — return all triples
   548→        patterns = [("?s", "?p", "?o")]
   549→        if not select_vars or select_vars == ["*"]:
   550→            select_vars = ["?s", "?p", "?o"]
   551→
   552→    # split patterns into fact-patterns vs edge-patterns so we
   553→    # search the right materialized triple list
   554→    fact_pats = [p for p in patterns if not is_edge_predicate(p[1])]
   555→    edge_pats = [p for p in patterns if is_edge_predicate(p[1])]
   556→    edge_pool = edge_triples or []
   557→
   558→    # naive nested-loop join: for each pattern, extend the binding set
   559→    bindings_list: list[dict] = [{}]
   560→    for pat in fact_pats:
   561→        new_list = []
   562→        for b in bindings_list:
   563→            for tri in facts:
   564→                nb = match_triple(tri, pat, b)
   565→                if nb is not None:
   566→                    new_list.append(nb)
   567→        bindings_list = new_list
   568→        if not bindings_list:
   569→            break
   570→    # edge patterns join against the edges materialized view
   571→    for pat in edge_pats:
   572→        new_list = []
   573→        for b in bindings_list:
   574→            for tri in edge_pool:
   575→                nb = match_triple(tri, pat, b)
   576→                if nb is not None:
   577→                    new_list.append(nb)
   578→        bindings_list = new_list
   579→        if not bindings_list:
   580→            break
   581→
   582→    # OPTIONAL: left-join each optional pattern group
   583→    for opt_pats in optionals:
   584→        new_list = []
   585→        for b in bindings_list:
   586→            extended = [b]
   587→            for op in opt_pats:
   588→                pool = edge_pool if is_edge_predicate(op[1]) else facts
   589→                next_ext = []
   590→                for bb in extended:
   591→                    for tri in pool:
   592→                        nb = match_triple(tri, op, bb)
   593→                        if nb is not None:
   594→                            next_ext.append(nb)
   595→                extended = next_ext if next_ext else [b]  # left-join: keep b
   596→            # add all extended bindings (or original if no match)
   597→            new_list.extend(extended)
   598→        bindings_list = new_list
   599→
   600→    # apply filters
   601→    if filters:
   602→        bindings_list = [b for b in bindings_list if apply_filters(b, filters)]
   603→
   604→    # blob resolution: if the SELECT projects ?o and ?o is a chunk_id
   605→    # in the arena, optionally resolve to actual text. We do this only
   606→    # if a resolver is provided AND a ?var in SELECT has chunk_id shape
   607→    # (32+ hex chars). Conservative — never resolves without a resolver.
   608→    if blob_resolver and select_vars:
   609→        for b in bindings_list:
   610→            for v in select_vars:
   611→                val = b.get(v)
   612→                if isinstance(val, str) and len(val) >= 32 and \
   613→                        re.fullmatch(r"[a-f0-9]+", val[:32]):
   614→                    try:
   615→                        resolved = blob_resolver(None, None, val)
   616→                        if resolved and len(resolved) > 0:
   617→                            b[v] = resolved
   618→                    except Exception:
   619→                        pass  # keep chunk_id on resolution failure
   620→
   621→    # ORDER BY
   622→    if order_by:
   623→        var = order_by["var"]
   624→        desc = order_by["desc"]
   625→        bindings_list.sort(
   626→            key=lambda b: (b.get(var) or ""),
   627→            reverse=desc)
   628→
   629→    # DISTINCT
   630→    if parsed["distinct"]:
   631→        seen: set = set()
   632→        deduped = []
   633→        for b in bindings_list:
   634→            key = tuple((v, b.get(v)) for v in select_vars)
   635→            if key not in seen:
   636→                seen.add(key)
   637→                deduped.append(b)
   638→        bindings_list = deduped
   639→
   640→    # OFFSET
   641→    if offset:
   642→        bindings_list = bindings_list[offset:]
   643→
   644→    # LIMIT
   645→    if limit is not None:
   646→        bindings_list = bindings_list[:limit]
   647→
   648→    # project to SELECT vars
   649→    rows = []
   650→    for b in bindings_list:
   651→        row = {v: b.get(v, "") for v in select_vars}
   652→        rows.append(row)
   653→
   654→    return {
   655→        "head": {"vars": select_vars},
   656→        "results": {"bindings": rows},
   657→        "n_results": len(rows),
   658→    }
   659→

   661→# ----------------------------------------------------------- HTTP server
   662→def make_handler(mem, user_id: str | None = None):
   663→    """Build a BaseHTTPRequestHandler bound to a Memory instance."""
   664→
   665→    class SparqlHandler(BaseHTTPRequestHandler):
   666→        def _send(self, code: int, body: str,
   667→                  content_type: str = "application/json"):
   668→            data = body.encode("utf-8")
   669→            self.send_response(code)
   670→            self.send_header("Content-Type", content_type)
   671→            self.send_header("Content-Length", str(len(data)))
   672→            self.send_header("Access-Control-Allow-Origin", "*")
   673→            self.send_header("Access-Control-Allow-Methods",
   674→                              "GET, POST, OPTIONS")
   675→            self.send_header("Access-Control-Allow-Headers",
   676→                              "Content-Type, Authorization")
   677→            self.end_headers()
   678→            try:
   679→                self.wfile.write(data)
   680→            except (BrokenPipeError, ConnectionResetError):
   681→                pass
   682→
   683→        def do_OPTIONS(self):
   684→            self._send(200, "{}")
   685→
   686→        def do_GET(self):
   687→            parsed_url = urllib.parse.urlparse(self.path)
   688→            params = urllib.parse.parse_qs(parsed_url.query)
   689→            query = params.get("query", [""])[0]
   690→            self._handle_query(query)
   691→
   692→        def do_POST(self):
   693→            length = int(self.headers.get("Content-Length", 0))
   694→            body = self.rfile.read(length).decode("utf-8") if length else ""
   695→            content_type = self.headers.get("Content-Type", "")
   696→            if "x-www-form-urlencoded" in content_type:
   697→                params = urllib.parse.parse_qs(body)
   698→                query = params.get("query", [""])[0]
   699→            else:
   700→                query = body
   701→            self._handle_query(query)
   702→
   703→        def _handle_query(self, query: str):
   704→            if not query:
   705→                self._send(400, json.dumps({
   706→                    "error": "missing 'query' parameter — send "
   707→                              "'?query=SELECT ...' or POST a SPARQL query"}))
   708→                return
   709→            try:
   710→                # pull facts + edges
   711→                fact_objs = mem.store.query_facts(active=True, user_id=user_id)
   712→                facts = [(f.subject, f.relation, f.value) for f in fact_objs]
   713→                edges = edge_triples(mem, user_id=user_id)
   714→                # build blob resolver if arena is enabled
   715→                blob_resolver = None
   716→                arena = getattr(mem, "blob_arena", None)
   717→                if arena is not None:
   718→                    def _resolve(_s, _r, val):
   719→                        from context_m.trace.blob_arena import get_chunk_text
   720→                        return get_chunk_text(mem.store, arena, val)
   721→                    blob_resolver = _resolve
   722→                result = execute_sparql(query, facts, user_id,
   723→                                         edge_triples=edges,
   724→                                         blob_resolver=blob_resolver)
   725→                self._send(200, json.dumps(result, indent=2, default=str))
   726→            except ValueError as e:
   727→                self._send(400, json.dumps({"error": str(e)}))
   728→            except Exception as e:  # noqa: BLE001
   729→                self._send(500, json.dumps({"error": f"server: {e}"}))
   730→
   731→        def log_message(self, fmt, *args):
   732→            pass  # quiet
   733→
   734→    return SparqlHandler

   736→
   737→class SparqlServer:
   738→    """A minimal SPARQL HTTP endpoint backed by a Memory instance.
   739→
   740→    The Memory's reader is configured with the RDFDecoder so retrieval
   741→    produces RDF triples instead of an LLM prompt. This server exposes
   742→    those triples via SPARQL — a fully non-LLM retrieval path.
   743→
   744→    Threaded: uses ThreadingHTTPServer so concurrent queries don't
   745→    block each other. Each thread reads from the shared Memory (which
   746→    is itself guarded by an RLock at the store level).
   747→    """
   748→    def __init__(self, mem, host: str = "127.0.0.1", port: int = 8910,
   749→                 user_id: str | None = None) -> None:
   750→        self.mem = mem
   751→        self.host = host
   752→        self.port = port
   753→        self.user_id = user_id
   754→        self._server: ThreadingHTTPServer | None = None
   755→        self._thread: threading.Thread | None = None
   756→
   757→    def start(self) -> None:
   758→        handler_cls = make_handler(self.mem, user_id=self.user_id)
   759→        self._server = ThreadingHTTPServer((self.host, self.port),
   760→                                            handler_cls)
   761→        self._server.daemon_threads = True
   762→        print(f"[sparql] listening on http://{self.host}:{self.port}/"
   763→              f"  (SPARQL 1.1 Protocol; thread-per-request)")
   764→        print(f"[sparql]   user_id: {self.user_id or '(all users)'}")
   765→        print(f"[sparql]   try: curl 'http://{self.host}:{self.port}/"
   766→              f"?query=SELECT%20%3Fs%20%3Fp%20%3Fo%20WHERE%20%7B%20%3Fs%20"
   767→              f"%3Fp%20%3Fo%20%7D%20LIMIT%2010'")
   768→
   769→    def start_background(self) -> None:
   770→        """Start the server in a daemon thread (non-blocking).
   771→
   772→        Used by `cortexm serve-rest --sparql-port N` to share one Memory
   773→        instance across both the REST API and the SPARQL endpoint.
   774→        """
   775→        self.start()
   776→        self._thread = threading.Thread(
   777→            target=self._server.serve_forever, daemon=True,
   778→            name=f"sparql-{self.port}")
   779→        self._thread.start()
   780→
   781→    def serve_forever(self) -> None:
   782→        if self._server is None:
   783→            self.start()
   784→        assert self._server is not None
   785→        self._server.serve_forever()
   786→
   787→    def stop(self) -> None:
   788→        if self._server is not None:
   789→            self._server.shutdown()
   790→            self._server.server_close()
   791→            self._server = None
   792→        if self._thread is not None:
   793→            self._thread.join(timeout=2)
   794→            self._thread = None
   795→
   796→
   797→def main() -> int:
   798→    import argparse
   799→    ap = argparse.ArgumentParser(prog="contextm-sparql",
   800→                                  description="Context-M SPARQL endpoint")
   801→    ap.add_argument("--host", default="127.0.0.1")
   802→    ap.add_argument("--port", type=int, default=8910)
   803→    ap.add_argument("--db", default=None,
   804→                    help="path to the Context-M DB (default: in-memory)")
   805→    ap.add_argument("--user-id", default=None,
   806→                    help="scope queries to a single user")
   807→    args = ap.parse_args()
   808→
   809→    from context_m.api.memory import Memory
   810→    from context_m.config import Config
   811→    cfg = Config.from_env()
   812→    if args.db:
   813→        cfg.db_path = args.db
   814→    mem = Memory(cfg)
   815→    server = SparqlServer(mem, host=args.host, port=args.port,
   816→                           user_id=args.user_id)
   817→    try:
   818→        server.serve_forever()
   819→    except KeyboardInterrupt:
   820→        print("\n[sparql] shutting down")
   821→    finally:
   822→        server.stop()
   823→        mem.close()
   824→    return 0
